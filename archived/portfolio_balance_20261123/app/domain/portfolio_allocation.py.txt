"""Allocation rules. Weights are fractions; signed shortfall is target minus current."""

from dataclasses import dataclass
from datetime import datetime, timezone
import math
from collections.abc import Mapping, Sequence

from app.domain.market.symbol_catalog import is_usd_equivalent_symbol


def nonnegative(value: object) -> float:
    return (
        float(value)
        if isinstance(value, (int, float)) and not isinstance(value, bool) and math.isfinite(value) and value >= 0
        else 0.0
    )


def normalized_weights(assets: Sequence[Mapping]) -> tuple[list[float], float, bool]:
    total = sum(nonnegative(asset.get("targetWeight")) for asset in assets)
    if not assets:
        return [], 0.0, False
    if total <= 0:
        return [1 / len(assets)] * len(assets), total, True
    return [nonnegative(asset.get("targetWeight")) / total for asset in assets], total, False


def usable_asset_price(asset: Mapping, *, now: datetime, require_fresh: bool) -> float | None:
    if is_usd_equivalent_symbol(str(asset.get("symbol") or "")):
        return 1.0
    price = nonnegative(asset.get("currentPrice"))
    if price <= 0:
        return None
    if require_fresh:
        try:
            expiry = datetime.fromisoformat(str(asset["priceValidUntil"]).replace("Z", "+00:00"))
            observed = asset.get("priceObservedAt")
            if expiry.tzinfo is None or expiry <= now:
                return None
            if observed:
                stamp = datetime.fromisoformat(str(observed).replace("Z", "+00:00"))
                if stamp.tzinfo is None or (stamp - now).total_seconds() > 5 or stamp >= expiry:
                    return None
        except (KeyError, TypeError, ValueError):
            return None
    return price


def money(value: float) -> float:
    # Same nonnegative half-up boundary used by the browser preview.
    return math.floor((value + 2.220446049250313e-16) * 100 + 0.5) / 100


@dataclass(frozen=True)
class Allocation:
    assets: list[Mapping]
    total_value: float
    target_weight_input_sum: float
    uses_equal_weight_fallback: bool
    valuation_complete: bool
    values: list[float]
    target_weights: list[float]
    current_weights: list[float]

    def shortfall_weight(self, index: int) -> float:
        return self.target_weights[index] - self.current_weights[index]

    def overweight_percentage_points(self, index: int) -> float | None:
        """Legacy signed alerts mean overweight; an unvalued portfolio has no drift."""
        if not self.valuation_complete or self.total_value <= 0:
            return None
        return -self.shortfall_weight(index) * 100


def calculate_allocation(assets: Sequence[Mapping], *, require_fresh: bool, now: datetime | None = None) -> Allocation:
    now = now or datetime.now(timezone.utc)
    assets = [
        asset
        for asset in assets
        if str(asset.get("symbol") or "").strip()
        or any(nonnegative(asset.get(key)) > 0 for key in ("targetWeight", "units", "currentPrice"))
    ]
    weights, weight_sum, fallback = normalized_weights(assets)
    prices = [usable_asset_price(asset, now=now, require_fresh=require_fresh) for asset in assets]
    complete = all(nonnegative(asset.get("units")) <= 0 or price is not None for asset, price in zip(assets, prices))
    values = [money(nonnegative(asset.get("units")) * (price or 0)) for asset, price in zip(assets, prices)]
    total = money(sum(values))
    return Allocation(
        assets,
        total,
        weight_sum,
        fallback,
        complete,
        values,
        weights,
        [value / total if total > 0 else 0.0 for value in values],
    )
