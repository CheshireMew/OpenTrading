from __future__ import annotations

from sqlalchemy import delete, func, select

from app.domain.portfolios import (
    PortfolioCollectionMismatchError,
    PortfolioCollectionRecord,
    PortfolioDeletedPageRecord,
    PortfolioDeletedSummaryRecord,
    PortfolioNotFoundError,
    PortfolioRevisionConflictError,
    portfolio_simulation_fingerprint,
)

from app.infra.db.database import DatabaseRuntime

from app.infra.persistence.write_session import write_session

from app.infra.db.schema import (
    PortfolioCollectionState,
    PortfolioRecord,
    PortfolioSimulationResultRecord as PortfolioSimulationResultRow,
)

from utils.time import utc_now_naive

from app.infra.persistence.portfolio_rows import COLLECTION_ID, claim_revision, to_record


class PortfolioCollectionRepository:
    def __init__(self, database_runtime: DatabaseRuntime) -> None:
        self._database_runtime = database_runtime

    def get_collection(self) -> PortfolioCollectionRecord:
        with write_session(
            self._database_runtime, PortfolioRevisionConflictError, PortfolioCollectionMismatchError
        ) as session:
            state = session.get(PortfolioCollectionState, COLLECTION_ID)
            revision = state.revision if state is not None else 0
            active_portfolio_id = state.active_portfolio_id if state is not None else ""
            records = list(
                session.scalars(
                    select(PortfolioRecord)
                    .where(PortfolioRecord.deleted_at.is_(None))
                    .order_by(PortfolioRecord.position, PortfolioRecord.created_at, PortfolioRecord.id)
                )
            )
            return to_record(revision=revision, active_portfolio_id=active_portfolio_id, records=records)

    def list_deleted(self, *, offset: int, limit: int) -> PortfolioDeletedPageRecord:
        with write_session(
            self._database_runtime, PortfolioRevisionConflictError, PortfolioCollectionMismatchError
        ) as session:
            total = int(
                session.scalar(
                    select(func.count()).select_from(PortfolioRecord).where(PortfolioRecord.deleted_at.is_not(None))
                )
                or 0
            )
            records = list(
                session.scalars(
                    select(PortfolioRecord)
                    .where(PortfolioRecord.deleted_at.is_not(None))
                    .order_by(PortfolioRecord.deleted_at.desc(), PortfolioRecord.id)
                    .offset(offset)
                    .limit(limit)
                )
            )
            return PortfolioDeletedPageRecord(
                items=[
                    PortfolioDeletedSummaryRecord(
                        portfolio_id=record.id,
                        name=str(record.payload.get("name") or record.id),
                        deleted_at=record.deleted_at,
                    )
                    for record in records
                ],
                offset=offset,
                limit=limit,
                total=total,
            )

    def apply_collection_changes(
        self,
        active_portfolio_id: str,
        changed_portfolio_payloads: list[dict],
        portfolio_order: list[str],
        expected_revision: int,
    ) -> PortfolioCollectionRecord:
        with write_session(
            self._database_runtime, PortfolioRevisionConflictError, PortfolioCollectionMismatchError
        ) as session:
            now = utc_now_naive()
            revision = claim_revision(
                session, expected_revision=expected_revision, active_portfolio_id=active_portfolio_id, updated_at=now
            )
            existing = {record.id: record for record in session.scalars(select(PortfolioRecord))}
            changed_by_id = {str(payload["id"]): payload for payload in changed_portfolio_payloads}
            ordered_ids = set(portfolio_order)
            positions = {portfolio_id: position for position, portfolio_id in enumerate(portfolio_order)}
            missing_ids = [
                portfolio_id
                for portfolio_id in portfolio_order
                if portfolio_id not in existing and portfolio_id not in changed_by_id
            ]
            unrestored_ids = [
                portfolio_id
                for portfolio_id in portfolio_order
                if portfolio_id in existing
                and existing[portfolio_id].deleted_at is not None
                and (portfolio_id not in changed_by_id)
            ]
            if missing_ids or unrestored_ids:
                invalid_ids = missing_ids + unrestored_ids
                raise PortfolioCollectionMismatchError(f"组合变更引用了不存在的活动组合: {', '.join(invalid_ids)}")
            for payload in changed_portfolio_payloads:
                portfolio_id = str(payload["id"])
                record = existing.get(portfolio_id)
                if record is None:
                    record = PortfolioRecord(
                        id=portfolio_id,
                        payload=payload,
                        version=1,
                        position=positions[portfolio_id],
                        deleted_at=None,
                        created_at=now,
                        updated_at=now,
                    )
                    session.add(record)
                    existing[portfolio_id] = record
                    continue
                changed = record.payload != payload or record.deleted_at is not None
                simulation_inputs_changed = portfolio_simulation_fingerprint(
                    record.payload
                ) != portfolio_simulation_fingerprint(payload)
                record.payload = payload
                record.deleted_at = None
                if changed:
                    record.version += 1
                    record.updated_at = now
                if simulation_inputs_changed:
                    session.execute(
                        delete(PortfolioSimulationResultRow).where(
                            PortfolioSimulationResultRow.portfolio_id == portfolio_id
                        )
                    )
            for portfolio_id, record in existing.items():
                if portfolio_id in ordered_ids:
                    position = positions[portfolio_id]
                    if record.position != position:
                        record.position = position
                        record.version += 1
                        record.updated_at = now
                    continue
                if record.deleted_at is not None:
                    continue
                record.deleted_at = now
                record.version += 1
                record.updated_at = now
            session.flush()
            records = list(
                session.scalars(
                    select(PortfolioRecord)
                    .where(PortfolioRecord.deleted_at.is_(None))
                    .order_by(PortfolioRecord.position, PortfolioRecord.created_at, PortfolioRecord.id)
                )
            )
            return to_record(revision=revision, active_portfolio_id=active_portfolio_id, records=records)

    def restore_portfolio(self, portfolio_id: str, expected_revision: int) -> PortfolioCollectionRecord:
        with write_session(
            self._database_runtime, PortfolioRevisionConflictError, PortfolioCollectionMismatchError
        ) as session:
            now = utc_now_naive()
            revision = claim_revision(
                session, expected_revision=expected_revision, active_portfolio_id=portfolio_id, updated_at=now
            )
            record = session.scalar(select(PortfolioRecord).where(PortfolioRecord.id == portfolio_id))
            if record is None or record.deleted_at is None:
                raise PortfolioNotFoundError("回收站中没有这个组合")
            max_position = max(
                (
                    item.position
                    for item in session.scalars(select(PortfolioRecord).where(PortfolioRecord.deleted_at.is_(None)))
                ),
                default=-1,
            )
            record.deleted_at = None
            record.position = max_position + 1
            record.version += 1
            record.updated_at = now
            session.flush()
            records = list(
                session.scalars(
                    select(PortfolioRecord).order_by(
                        PortfolioRecord.deleted_at.is_not(None),
                        PortfolioRecord.position,
                        PortfolioRecord.created_at,
                        PortfolioRecord.id,
                    )
                )
            )
            return to_record(revision=revision, active_portfolio_id=record.id, records=records)
