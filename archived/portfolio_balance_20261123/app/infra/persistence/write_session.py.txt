"""Translate database-enforced write conflicts without leaking storage errors."""
from contextlib import contextmanager

from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm.exc import StaleDataError


@contextmanager
def write_session(database, conflict_error, duplicate_error=None):
    try:
        with database.session_scope() as session:
            yield session
    except StaleDataError as exc:
        raise conflict_error("记录已在其它窗口更新，请重新载入后再试") from exc
    except IntegrityError as exc:
        # Only translate the ledger's known uniqueness rule. Other integrity
        # failures must retain their own diagnostics and transaction rollback.
        detail = str(exc.orig)
        if duplicate_error is not None and (
            "uq_portfolio_active_external_id" in detail
            or "UNIQUE constraint failed: portfolio_transactions.portfolio_id, portfolio_transactions.external_id" in detail
        ):
            raise duplicate_error("这个外部流水编号已经存在，请勿重复入账") from exc
        raise
