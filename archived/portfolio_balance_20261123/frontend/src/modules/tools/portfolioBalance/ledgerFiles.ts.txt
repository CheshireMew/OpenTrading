import type {
  PortfolioTransactionInput,
  PortfolioTransactionResponse,
} from '@/modules/tools/contracts'

const HEADERS = [
  'transactionType',
  'symbol',
  'quantity',
  'unitPrice',
  'cashAmount',
  'fee',
  'currency',
  'feeCurrency',
  'externalId',
  'note',
  'occurredAt',
] as const

export class LedgerFileError extends Error {
  constructor(public readonly code: string, public readonly details: Record<string, string | number> = {}) {
    super(`Invalid ledger file: ${code}`)
    this.name = 'LedgerFileError'
  }
}

const parseCsv = (text: string): string[][] => {
  const rows: string[][] = []
  let row: string[] = []
  let value = ''
  let quoted = false
  for (let index = 0; index < text.length; index += 1) {
    const char = text[index]
    if (quoted) {
      if (char === '"' && text[index + 1] === '"') {
        value += '"'
        index += 1
      } else if (char === '"') {
        quoted = false
      } else {
        value += char
      }
      continue
    }
    if (char === '"') quoted = true
    else if (char === ',') {
      row.push(value)
      value = ''
    } else if (char === '\n') {
      row.push(value.replace(/\r$/, ''))
      if (row.some((cell) => cell.trim())) rows.push(row)
      row = []
      value = ''
    } else value += char
  }
  row.push(value.replace(/\r$/, ''))
  if (quoted) throw new LedgerFileError('csvQuotes')
  if (row.some((cell) => cell.trim())) rows.push(row)
  return rows
}

const numberValue = (value: unknown, field: string) => {
  const parsed = Number(value || 0)
  if (!Number.isFinite(parsed) || parsed < 0) throw new LedgerFileError('number', { field })
  return parsed
}

const normalize = (value: Record<string, unknown>): PortfolioTransactionInput => {
  if (!value || typeof value !== 'object' || Array.isArray(value)) throw new LedgerFileError('row')
  const transactionType = String(value.transactionType || '').trim() as PortfolioTransactionInput['transactionType']
  if (!['buy', 'sell', 'deposit', 'withdrawal', 'fee', 'transfer_in', 'transfer_out'].includes(transactionType)) {
    throw new LedgerFileError('transactionType')
  }
  const occurredAt = String(value.occurredAt || '').trim()
  if (!occurredAt || Number.isNaN(new Date(occurredAt).getTime())) throw new LedgerFileError('occurredAt')
  return {
    transactionType,
    symbol: String(value.symbol || '').trim(),
    quantity: numberValue(value.quantity, 'quantity'),
    unitPrice: numberValue(value.unitPrice, 'unitPrice'),
    cashAmount: numberValue(value.cashAmount, 'cashAmount'),
    fee: numberValue(value.fee, 'fee'),
    currency: String(value.currency || 'USD').trim().toUpperCase(),
    feeCurrency: String(value.feeCurrency || value.currency || 'USD').trim().toUpperCase(),
    externalId: String(value.externalId || '').trim() || null,
    note: String(value.note || ''),
    metadata: {},
    occurredAt: new Date(occurredAt).toISOString(),
  }
}

export const parsePortfolioLedgerFile = async (file: File): Promise<PortfolioTransactionInput[]> => {
  const text = await file.text()
  if (file.name.toLowerCase().endsWith('.json')) {
    let parsed: unknown
    try { parsed = JSON.parse(text) } catch { throw new LedgerFileError('jsonSyntax') }
    const rows = Array.isArray(parsed)
      ? parsed
      : parsed && typeof parsed === 'object' && Array.isArray((parsed as { transactions?: unknown }).transactions)
        ? (parsed as { transactions: unknown[] }).transactions
        : null
    if (!rows) throw new LedgerFileError('jsonShape')
    return rows.map((row, index) => normalizeRow(row, index + 1))
  }
  const [headerRow, ...rows] = parseCsv(text)
  if (!headerRow) throw new LedgerFileError('empty')
  const header = headerRow.map((value) => value.trim())
  if (!header.includes('transactionType') || !header.includes('occurredAt')) throw new LedgerFileError('csvHeaders')
  return rows.map((cells, index) => normalizeRow(Object.fromEntries(
    header.map((key, index) => [key, cells[index] || '']),
  ), index + 2))
}

const normalizeRow = (row: unknown, index: number) => {
  try { return normalize(row as Record<string, unknown>) }
  catch (cause) {
    if (cause instanceof LedgerFileError) throw new LedgerFileError(cause.code, { ...cause.details, row: index })
    throw cause
  }
}

const escapeCsv = (value: unknown) => {
  const text = String(value ?? '')
  return /[",\r\n]/.test(text) ? `"${text.replace(/"/g, '""')}"` : text
}

export const portfolioLedgerCsv = (rows: PortfolioTransactionResponse[]) => [
  HEADERS.join(','),
  ...rows.map((row) => HEADERS.map((key) => escapeCsv(row[key])).join(',')),
].join('\r\n')

export const portfolioLedgerJson = (rows: PortfolioTransactionResponse[]) => JSON.stringify({
  schemaVersion: 1,
  exportedAt: new Date().toISOString(),
  transactions: rows.map(({ id: _id, portfolioId: _portfolioId, version: _version, archivedAt: _archivedAt, createdAt: _createdAt, updatedAt: _updatedAt, ...row }) => row),
}, null, 2)
