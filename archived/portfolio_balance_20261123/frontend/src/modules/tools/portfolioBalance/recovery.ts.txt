import type { PortfolioBalanceSnapshot } from './types'

export interface PortfolioRecovery {
  snapshot: PortfolioBalanceSnapshot
  baseline: PortfolioBalanceSnapshot
  revision: number
  pending: boolean
}

let database: Promise<IDBDatabase> | undefined
const windowKey = crypto.randomUUID()
const previousWindowKey = (() => {
  try {
    const previous = sessionStorage.getItem('opentrading-portfolio-window')
    sessionStorage.setItem('opentrading-portfolio-window', windowKey)
    return previous
  } catch { return null }
})()
const open = () => database ??= new Promise<IDBDatabase>((resolve, reject) => {
  const request = indexedDB.open('opentrading-portfolio-recovery', 1)
  request.onupgradeneeded = () => request.result.createObjectStore('drafts')
  request.onsuccess = () => resolve(request.result)
  request.onerror = () => { database = undefined; reject(request.error) }
})
export async function readPortfolioRecovery(): Promise<PortfolioRecovery | undefined> {
  const db = await open()
  return new Promise((resolve, reject) => {
    const store = db.transaction('drafts').objectStore('drafts')
    const request = store.get(windowKey)
    request.onsuccess = () => {
      if (request.result || !previousWindowKey) { resolve(request.result); return }
      const previous = store.get(previousWindowKey)
      previous.onsuccess = () => resolve(previous.result)
      previous.onerror = () => reject(previous.error)
    }
    request.onerror = () => reject(request.error)
  })
}
export async function writePortfolioRecovery(value: PortfolioRecovery) {
  const db = await open()
  const snapshot = JSON.parse(JSON.stringify(value))
  return new Promise<void>((resolve, reject) => {
    const transaction = db.transaction('drafts', 'readwrite')
    transaction.objectStore('drafts').put(snapshot, windowKey)
    transaction.oncomplete = () => resolve()
    transaction.onerror = () => reject(transaction.error)
    transaction.onabort = () => reject(transaction.error)
  })
}

const same = (left: unknown, right: unknown) => JSON.stringify(left) === JSON.stringify(right)
export function mergePortfolioRecovery(local: PortfolioBalanceSnapshot, baseline: PortfolioBalanceSnapshot, remote: PortfolioBalanceSnapshot) {
  const old = new Map(baseline.portfolios.map(item => [item.id, item]))
  const mine = new Map(local.portfolios.map(item => [item.id, item]))
  const theirs = new Map(remote.portfolios.map(item => [item.id, item]))
  const merged = new Map(theirs)
  const conflicts: string[] = []
  for (const id of new Set([...old.keys(), ...mine.keys()])) {
    if (same(mine.get(id), old.get(id))) continue
    if (!same(theirs.get(id), old.get(id)) && !same(theirs.get(id), mine.get(id))) {
      conflicts.push(id)
      continue
    }
    const value = mine.get(id)
    if (value) merged.set(id, value)
    else merged.delete(id)
  }
  const localReordered = !same(local.portfolios.map(item => item.id), baseline.portfolios.map(item => item.id))
  if (localReordered && !same(remote.portfolios.map(item => item.id), baseline.portfolios.map(item => item.id))
    && !same(local.portfolios.map(item => item.id), remote.portfolios.map(item => item.id))) conflicts.push('order')
  const order = [...(localReordered ? local : remote).portfolios.map(item => item.id), ...merged.keys()]
  const portfolios = [...new Set(order)].flatMap(id => merged.has(id) ? [merged.get(id)!] : [])
  const preferred = local.activePortfolioId !== baseline.activePortfolioId ? local.activePortfolioId : remote.activePortfolioId
  return { conflicts, snapshot: { portfolios, activePortfolioId: portfolios.some(item => item.id === preferred) ? preferred : portfolios[0]?.id || '' } }
}
