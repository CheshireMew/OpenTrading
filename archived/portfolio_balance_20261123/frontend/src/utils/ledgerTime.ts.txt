// The API stores UTC. A datetime-local field is interpreted in the selected display zone.
export function utcInstant(value: string | Date): Date {
  const normalized = typeof value === 'string' && !/(?:Z|[+-]\d{2}:?\d{2})$/i.test(value) ? `${value}Z` : value
  const date = normalized instanceof Date ? new Date(normalized.getTime()) : new Date(normalized)
  if (!Number.isFinite(date.getTime())) throw new Error('Invalid transaction time')
  return date
}

export function ledgerTimeInput(value: string | Date, zone: string): string {
  const date = utcInstant(value)
  const parts = new Intl.DateTimeFormat('en-CA', {
    timeZone: zone, year: 'numeric', month: '2-digit', day: '2-digit',
    hour: '2-digit', minute: '2-digit', second: '2-digit', hourCycle: 'h23',
  }).formatToParts(date)
  const part = (type: string) => parts.find(item => item.type === type)!.value
  return `${part('year')}-${part('month')}-${part('day')}T${part('hour')}:${part('minute')}:${part('second')}.${String(date.getUTCMilliseconds()).padStart(3, '0')}`
}

export function ledgerTimeInstant(input: string, zone: string, original?: string): string {
  // Keep the API's sub-millisecond precision when the user did not edit time.
  if (original && ledgerTimeInput(original, zone) === input) return /(?:Z|[+-]\d{2}:?\d{2})$/i.test(original) ? original : `${original}Z`
  const wall = new Date(`${input}Z`)
  if (!Number.isFinite(wall.getTime())) throw new Error('Invalid transaction time')
  const canonical = wall.toISOString().slice(0, -1)
  const offsets = new Set<number>()
  for (let hours = -36; hours <= 36; hours += 3) {
    const probe = new Date(wall.getTime() + hours * 3600000)
    offsets.add(new Date(`${ledgerTimeInput(probe, zone)}Z`).getTime() - probe.getTime())
  }
  const matches = [...offsets].map(offset => new Date(wall.getTime() - offset))
    .filter(candidate => ledgerTimeInput(candidate, zone) === canonical)
  if (matches.length !== 1) throw new Error('该时刻处于夏令时切换的缺失或重复区间，请选择明确时刻或切换到 UTC 后输入')
  return matches[0]!.toISOString()
}
