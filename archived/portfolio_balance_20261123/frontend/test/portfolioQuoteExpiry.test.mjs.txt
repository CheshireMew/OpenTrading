import assert from 'node:assert/strict'
import fs from 'node:fs'
import test from 'node:test'
import * as vue from 'vue'
import ts from 'typescript'

function load(name, modules) {
  const exports = {}
  const code = ts.transpileModule(fs.readFileSync(new URL('../src/modules/tools/portfolioBalance/' + name + '.ts', import.meta.url), 'utf8'),
    { compilerOptions: { module: ts.ModuleKind.CommonJS } }).outputText
  new Function('require', 'exports', code)(id => modules[id], exports)
  return exports
}

test('portfolio plan expires even when no network response changes the assets', async () => {
  let clock = 1000, timer, unmount
  const state = { valuationTime: vue.ref(0), portfolios: vue.ref([]), activePortfolioId: vue.ref('p'),
    activePortfolio: vue.ref({ id: 'p', assets: [] }), storageReady: vue.ref(false),
    clearSimulationResult() {}, assets: vue.ref([{ priceValidUntil: new Date(2000).toISOString() }]) }
  const { usePortfolioBalancePage } = load('usePortfolioBalancePage', {
    vue: { ...vue, onBeforeUnmount: callback => { unmount = callback } },
    'vue-i18n': { useI18n: () => ({ t: value => value }) },
    '@/modules/market': { toBaseSymbol: value => value },
    './state': { createPortfolioBalancePageState: () => state },
    './persistence': { createPortfolioPersistence: () => ({ initialize() {}, stop: async () => {}, queueSave() {} }) },
    './commands': { createPortfolioBalancePageCommands: () => ({ clearScheduledMarketRefresh() {} }) },
    './simulation': { portfolioSimulationInputKey: () => '' }, './snapshot': { hasMarketGap: () => false },
  })
  const realNow = Date.now, realTimeout = globalThis.setTimeout, realClear = globalThis.clearTimeout
  const realInterval = globalThis.setInterval, realClearInterval = globalThis.clearInterval
  try {
    Date.now = () => clock
    globalThis.setTimeout = (callback, delay) => { timer = { callback, delay }; return 1 }
    globalThis.clearTimeout = () => {}
    globalThis.setInterval = () => 2
    globalThis.clearInterval = () => {}
    const scope = vue.effectScope()
    scope.run(usePortfolioBalancePage)
    assert.equal(timer.delay, 1001)
    assert.equal(state.valuationTime.value, 1000)
    clock = 2001
    timer.callback()
    assert.equal(state.valuationTime.value, 2001)
    unmount()
    scope.stop()
  } finally {
    Date.now = realNow
    globalThis.setTimeout = realTimeout; globalThis.clearTimeout = realClear
    globalThis.setInterval = realInterval; globalThis.clearInterval = realClearInterval
  }
})
