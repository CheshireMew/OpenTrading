import time

from types import SimpleNamespace

import pytest
from uuid import uuid4

from app.services.market.price_query_service import MarketPriceQueryService
from app.services.market.binance_market_snapshot_service import BinanceMarketSnapshotService


@pytest.mark.asyncio
async def test_stale_candle_is_not_a_current_quote_or_given_query_timestamp():
    service = MarketPriceQueryService(market_data_service=SimpleNamespace(get_history_data=lambda *args: [[1704067200000, 100, 100, 100, 100, 1]]))
    result = await service.get_current_price(symbol="BTC/USDT", timeframe="1d")
    assert result.current_price is None
    assert result.status == "stale"
    assert result.timestamp.startswith("2024-01-01")
    assert result.queried_at != result.timestamp


@pytest.mark.asyncio
async def test_stale_streams_fail_health_and_never_mix_futures_into_spot(monkeypatch):
    service = BinanceMarketSnapshotService()
    service._running = service._seed_completed = True
    service._seed_success_count = 1
    now = int(time.time() * 1000)
    await service._store.merge_tickers("spot_ticker", [{"symbol": "BTCUSDT", "last_price": 100, "close_time": now}])
    await service._store.merge_tickers("usdm_ticker", [{"symbol": "BTCUSDT", "last_price": 110, "close_time": now}])
    await service._store.merge_marks([{"symbol": "BTCUSDT", "mark_price": 111, "time": now}])
    assert await service.get_current_price("BTC/USDT") == 100
    assert await service.get_current_price("binance:usdm:BTCUSDT") == 111
    assert service.health_snapshot()["status"] == "ready"
    monkeypatch.setattr("app.services.market.binance_market_snapshot_store.time.time", lambda: now / 1000 + 180)
    assert service.health_snapshot()["status"] == "stale"
    assert await service.get_current_price("BTC/USDT") is None
    snapshot = await service.get_market_page_snapshot_data()
    assert snapshot["load_errors"]
    assert snapshot["spot_ticker"]["items"] == []


def test_expired_saved_portfolio_quote_invalidates_valuation(installed_database_runtime):
    from app.infra.persistence.portfolio_repository import PortfolioRepository
    from app.services.tools.portfolio_service import PortfolioService
    from app.contracts.portfolios import PortfolioCollectionSaveCommand, PortfolioBalanceSnapshot, PortfolioTransactionCreateCommand
    from test.test_portfolio_persistence import _portfolio
    portfolio = _portfolio()
    portfolio.assets[0].priceObservedAt = (datetime.now(timezone.utc) - timedelta(days=1)).isoformat()
    portfolio.assets[0].priceValidUntil = (datetime.now(timezone.utc) - timedelta(hours=23)).isoformat()
    service = PortfolioService(PortfolioRepository(installed_database_runtime))
    service.save_collection(PortfolioCollectionSaveCommand(expectedRevision=0, snapshot=PortfolioBalanceSnapshot(activePortfolioId="core", portfolios=[portfolio])))
    service.create_transaction("core", PortfolioTransactionCreateCommand(requestId=uuid4().hex, transactionType="transfer_in", symbol="BTC", quantity=1, unitPrice=100, occurredAt="2026-01-01T00:00:00Z"))
    assert service.get_ledger("core").summary.totalEquity is None


@pytest.mark.asyncio
async def test_ledger_and_drift_use_backend_quotes_and_transactions_without_browser_saves(installed_database_runtime):
    from unittest.mock import AsyncMock
    from app.infra.persistence.portfolio_repository import PortfolioRepository
    from app.services.tools.portfolio_service import PortfolioService
    from app.services.research_alert_runtime import ResearchAlertRuntime
    from app.contracts.portfolios import PortfolioCollectionSaveCommand, PortfolioBalanceSnapshot, PortfolioTransactionCreateCommand
    from test.test_portfolio_persistence import _portfolio
    portfolio = _portfolio()
    portfolio.assets[0].units = 999  # Stale browser projection.
    portfolio.assets[0].targetWeight = 50
    now = datetime.now(timezone.utc)
    query = SimpleNamespace(get_current_price=AsyncMock(return_value=SimpleNamespace(
        current_price=200, status="fresh", timestamp=now.isoformat(), valid_until=(now + timedelta(seconds=120)).isoformat())))
    service = PortfolioService(PortfolioRepository(installed_database_runtime), market_query_service=query)
    service.save_collection(PortfolioCollectionSaveCommand(expectedRevision=0, snapshot=PortfolioBalanceSnapshot(activePortfolioId="core", portfolios=[portfolio])))
    for kind, fields in [("deposit", {"cashAmount": 200}), ("buy", {"symbol": "BTC", "quantity": 1, "unitPrice": 100})]:
        service.create_transaction("core", PortfolioTransactionCreateCommand(requestId=kind, transactionType=kind,
            occurredAt="2026-01-01T00:00:00Z" if kind == "deposit" else "2026-01-02T00:00:00Z", **fields))
    ledger = await service.get_valued_ledger("core")
    assert ledger.summary.totalEquity == 300
    assert ledger.summary.positions[0].units == 1
    collection = await service.get_allocation_collection()
    assert {item["symbol"]: item["units"] for item in collection.active_payloads[0]["assets"]} == {"BTC": 1, "USD": 100}
    drift = ResearchAlertRuntime._portfolio_drift({"symbol": "BTC", "configuration": {"portfolioId": "core"}}, collection)
    assert drift == pytest.approx(100 / 3)  # BTC is 2/3 of equity vs normalized 100% target.
    assert service.get_collection().revision == 1
    assert service.get_collection().snapshot.portfolios[0].assets[0].currentPrice == 0
    query.get_current_price.return_value.status = "stale"
    assert (await service.get_valued_ledger("core")).summary.totalEquity is None
    assert ResearchAlertRuntime._portfolio_drift({"symbol": "BTC", "configuration": {}}, await service.get_allocation_collection()) is None
