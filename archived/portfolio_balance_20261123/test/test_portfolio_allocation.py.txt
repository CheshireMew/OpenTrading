import json
from datetime import datetime
from pathlib import Path
from types import SimpleNamespace

import pytest

from app.domain.portfolio_allocation import calculate_allocation
from app.services.research_alert_runtime import ResearchAlertRuntime


CASES = json.loads((Path(__file__).parent / "fixtures/portfolio_allocation.json").read_text(encoding="utf-8"))


@pytest.mark.parametrize("case", CASES, ids=lambda case: case["name"])
def test_allocation_shared_browser_cases(case):
    allocation = calculate_allocation(
        case["assets"], require_fresh=case["requireFresh"], now=datetime.fromisoformat(case["now"])
    )
    assert allocation.total_value == pytest.approx(case["totalValue"])
    assert allocation.target_weights == pytest.approx(case["targetWeights"])
    assert allocation.current_weights == pytest.approx(case["currentWeights"])
    assert allocation.valuation_complete == case["complete"]
    assert allocation.uses_equal_weight_fallback == case["equalWeight"]


def test_alerts_use_equal_weights_and_reject_expired_and_missing_prices():
    assets = [
        {"symbol": "BTC", "units": 8, "currentPrice": 10, "targetWeight": 0, "priceValidUntil": "2099-01-01T00:00:00Z"},
        {"symbol": "ETH", "units": 2, "currentPrice": 10, "targetWeight": 0, "priceValidUntil": "2099-01-01T00:00:00Z"},
    ]
    collection = SimpleNamespace(active_portfolio_id="p", active_payloads=[{"id": "p", "assets": assets}])
    rule = {"symbol": "BTC/USDT", "configuration": {"signed": True}}
    assert ResearchAlertRuntime._portfolio_drift(rule, collection) == pytest.approx(30)
    assets[1]["priceValidUntil"] = "2000-01-01T00:00:00Z"
    assert ResearchAlertRuntime._portfolio_drift(rule, collection) is None
    assets[1]["priceValidUntil"] = "2099-01-01T00:00:00Z"
    assets[1]["currentPrice"] = 0
    assert ResearchAlertRuntime._portfolio_drift(rule, collection) is None
