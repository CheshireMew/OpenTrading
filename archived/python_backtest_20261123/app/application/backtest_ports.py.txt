"""Public capabilities used by research orchestration, independent of storage layout."""

from pathlib import Path
from typing import Any, Protocol

PreparedBacktest = tuple[dict[str, Any], bytes, dict[str, Any]]


class BacktestNotFoundError(ValueError):
    """The referenced run is absent; its saved research summary may still exist."""


class BacktestQueryPort(Protocol):
    def get(self, identifier: str) -> dict[str, Any]: ...
    def listing(self, offset: int = 0, limit: int = 50, archived: bool = False) -> dict[str, Any]: ...
    def records(
        self, identifier: str, kind: str, offset: int, limit: int, start: int | None = None, end: int | None = None
    ) -> dict[str, Any]: ...
    def artifacts(self, identifier: str) -> dict[str, Any]: ...


class ResearchBacktestPort(BacktestQueryPort, Protocol):
    @property
    def result_index_path(self) -> Path: ...
    def prepare(self, command: Any) -> PreparedBacktest: ...
    def preview_prepared(self, config: dict, metadata: dict, cache: dict | None = None) -> dict: ...
    def strategy_source(self, identifier: str) -> bytes: ...
    def save_execution_snapshot(self, metadata: dict, destination: Path, source_run: str | None = None) -> None: ...
    def create_from_snapshot(
        self,
        config: dict,
        source: bytes,
        metadata: dict,
        *,
        identifier: str | None = None,
        retry_from: str | None = None,
        execution_source: Path | None = None,
    ) -> dict: ...
    def verify_run(self, run: dict) -> dict: ...
    def resume(self, identifier: str, *, new_identifier: str | None = None) -> dict: ...
    def cancel(self, identifier: str) -> dict: ...
