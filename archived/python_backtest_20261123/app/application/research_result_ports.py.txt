from contextlib import AbstractContextManager
from pathlib import Path
from typing import Protocol


class ResultIndexTransactionPort(Protocol):
    def put(self, entry: dict) -> None: ...
    def get(self, identifier: str, source_type: str | None = None) -> dict | None: ...
    def entries(self, filters: dict) -> list[dict]: ...
    def discard(self, identifier: str, *, legacy_only: bool = False, unreferenced_only: bool = False) -> None: ...
    def source_results(self, source: str) -> list[str]: ...
    def references(self, identifier: str, exclude_source: str | None = None) -> list[tuple[str, str, str]]: ...
    def source_result(self, source: str, pointer: str) -> str | None: ...
    def forget_source(self, source: str) -> None: ...
    def forget_result_references(self, identifier: str) -> None: ...
    def link(self, source: str, pointer: str, identifier: str, digest: str) -> None: ...
    def import_status(self, source: str) -> tuple[str, str | None] | None: ...
    def mark_import(self, source: str, digest: str, count: int, issue: str | None) -> None: ...
    def imported_paths(self) -> list[str]: ...
    def annotated_sources(self) -> list[str]: ...


class ResultIndexPort(Protocol):
    path: Path

    def initialize(self) -> None: ...
    def transaction(self) -> AbstractContextManager[ResultIndexTransactionPort]: ...
