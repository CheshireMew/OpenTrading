from dataclasses import asdict
from math import isfinite
from types import MappingProxyType
from app.domain.backtest_types import BarResult, Context, Intent
from app.domain.backtest_validation import (
    validate_stop_entry,
    validate_stop_confirmation,
    observation_record,
    validate_risk_intent,
)


class StrategyDecisions:
    def __init__(self, state, records, config, history, params):
        self.state, self.records, self.config = state, records, config
        self.history, self.params = history, params

    def decide(self, strategy, bar, current):
        try:
            last_exit = (
                MappingProxyType(
                    {
                        k: self.records.trades[-1][k]
                        for k in ("trade_id", "side", "entry_time", "exit_time", "exit_reason", "return_pct")
                    }
                )
                if self.records.trades
                else None
            )
            intent = strategy.on_bar(
                Context(
                    self.history,
                    MappingProxyType(dict(self.state.position)) if self.state.position else None,
                    self.state.cash,
                    current,
                    self.params,
                    last_exit,
                )
            )
            entry_order = None
            self.state.trailing_distance = None
            if isinstance(intent, BarResult):
                distance = intent.trailing_distance
                if distance is not None and (isinstance(distance, bool) or not isinstance(distance, (int, float)) or not isfinite(distance) or distance <= 0):
                    raise ValueError("跟踪距离必须是有限正数")
                self.state.trailing_distance = distance
                if intent.record_observation:
                    self.records.observations.append(observation_record(intent, bar.timestamp))
                entry_order = intent.entry_order
                intent = intent.intent
            if intent is None:
                intent = Intent()
            validate_risk_intent(intent)
            validate_stop_confirmation(intent)
            if entry_order is not None:
                validate_stop_entry(entry_order)
                if intent.action not in ("hold", "close"):
                    raise ValueError("盘中入场委托不能和下一开盘入场同时发出")
        except Exception as exc:
            raise ValueError(
                f"策略 {getattr(strategy, '__file__', type(strategy).__name__)}，K 线 {bar.timestamp}: {exc}"
            ) from exc
        return intent, entry_order

    def schedule(self, intent, entry_order, bar):
        if intent.action != "hold":
            intent_fields = asdict(intent)
            if intent.profit_protection is None:
                intent_fields.pop("profit_protection")
            if intent.stop_loss_distance is None:
                intent_fields.pop("stop_loss_distance")
            if intent.activation_distance is None:
                intent_fields.pop("activation_distance")
            if intent.stop_loss_confirmation == "immediate":
                intent_fields.pop("stop_loss_confirmation")
            if not intent.pause_after_stop:
                intent_fields.pop("pause_after_stop")
            if intent.stop_confirmation_clock == "signal":
                intent_fields.pop("stop_confirmation_clock")
            signal = {
                "timestamp": bar.timestamp,
                "decision_time": bar.timestamp + self.config.step_ms,
                **intent_fields,
                "status": "pending",
                "signal_id": len(self.records.signals) + 1,
            }
            self.records.signals.append(signal)
            self.records.event(
                "signal",
                signal["decision_time"],
                signal_id=signal["signal_id"],
                action=signal["action"],
                reason=intent.reason,
            )
            self.state.pending = (signal, intent)
        if entry_order is not None:
            order_fields = asdict(entry_order.intent)
            if entry_order.intent.activation_distance is None:
                order_fields.pop("activation_distance")
            if entry_order.intent.stop_loss_confirmation == "immediate":
                order_fields.pop("stop_loss_confirmation")
            if not entry_order.intent.pause_after_stop:
                order_fields.pop("pause_after_stop")
            if entry_order.intent.stop_confirmation_clock == "signal":
                order_fields.pop("stop_confirmation_clock")
            signal = {
                "timestamp": bar.timestamp,
                "decision_time": bar.timestamp + self.config.step_ms,
                **order_fields,
                "order_type": "stop_entry",
                "trigger_price": entry_order.trigger_price,
                "worst_price": entry_order.worst_price,
                "ambiguity_policy": entry_order.ambiguity_policy,
                "status": "pending",
                "signal_id": len(self.records.signals) + 1,
            }
            self.records.signals.append(signal)
            self.records.event(
                "stop_entry_order",
                signal["decision_time"],
                signal_id=signal["signal_id"],
                action=signal["action"],
                trigger_price=entry_order.trigger_price,
                worst_price=entry_order.worst_price,
                ambiguity_policy=entry_order.ambiguity_policy,
            )
            self.state.pending_entry = (signal, entry_order)
