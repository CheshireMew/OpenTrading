from math import isfinite
from types import MappingProxyType
from app.domain.backtest_types import History
from app.domain.backtest_state import ExecutionState, SimulationConfig, BacktestRecords
from app.domain.backtest_account import BacktestAccount
from app.domain.backtest_orders import OrderExecution
from app.domain.backtest_protection import ProtectionExecution
from app.domain.backtest_report import build_report
from app.domain.backtest_decisions import StrategyDecisions


class Simulation:
    def __init__(self, config, parameters, record_factory=None):
        self.config = config
        self.state = ExecutionState(cash=config.initial_cash, resolution_ms=config.step_ms)
        self.records = BacktestRecords(config.initial_cash, record_factory)
        self.account = BacktestAccount(self.state, self.records, config)
        self.orders = OrderExecution(self.state, self.records, config, self.account)
        self.protection = ProtectionExecution(self.state, self.records, config)
        self.past_bars = []
        self.history = History(self.past_bars)
        self.params = MappingProxyType(dict(parameters))
        self.decisions = StrategyDecisions(self.state, self.records, config, self.history, self.params)

    def prepare_bar(self, bar):
        self.state.signal_bar_time = bar.timestamp
        if not self.records.equity:
            self.records.equity.append(
                {
                    "timestamp": bar.timestamp,
                    "equity": self.state.cash,
                    "cash": self.state.cash,
                    "unrealized": 0.0,
                    "drawdown_pct": 0.0,
                }
            )
        detail = self.config.execution_bars.get(bar.timestamp) if self.config.execution_bars is not None else None
        self.state.resolution_ms = (
            self.config.execution_step_ms or self.config.step_ms // len(detail) if detail else self.config.step_ms
        )
        self.state.fallback_time = bar.timestamp if self.config.execution_bars is not None and (not detail) else None
        if self.state.position and self.state.fallback_time is not None:
            self.state.position["execution_fallback_times"].append(self.state.fallback_time)
        if self.config.execution_bars is not None:
            if detail:
                self.records.detailed_candles += 1
            else:
                self.records.fallback_candles += 1
            if not self.records.execution_equity:
                self.records.execution_equity.append(
                    {
                        **self.records.equity[0],
                        "resolution_ms": self.state.resolution_ms,
                        "fallback": self.state.fallback_time is not None,
                    }
                )
        return detail

    def execute_subbar(self, sub, active_entry, detail):
        self.account.record_levels(sub)
        self.orders.thresholds(sub, gap=True)
        self.orders.process_pending(sub)
        active_entry = self.orders.check_entry(active_entry, sub)
        self.account.record_levels(sub)
        self.orders.thresholds(sub)
        if self.state.position and self.state.position.get("stop_confirmation_clock") == "execution":
            self.orders.finish_stop_confirmation(sub, self.state.resolution_ms)
        if detail and self.config.intrabar_profit_protection:
            self.protection.update(sub)
        if self.config.execution_bars is not None:
            marked = self.account.equity(sub.close)
            if not isfinite(marked):
                raise ValueError(f"账户数值溢出：{sub.timestamp}")
            self.records.execution_peak = max(self.records.execution_peak, marked)
            self.records.execution_equity.append(
                {
                    "timestamp": sub.timestamp + self.state.resolution_ms,
                    "equity": marked,
                    "cash": self.state.cash,
                    "unrealized": marked
                    - self.state.cash
                    - (self.state.position["principal"] if self.state.position else 0),
                    "drawdown_pct": (self.records.execution_peak - marked) / self.records.execution_peak * 100,
                    "resolution_ms": self.state.resolution_ms,
                    "fallback": self.state.fallback_time is not None,
                }
            )
        return active_entry

    def finish_bar(self, bar):
        if self.state.position and self.state.position.get("stop_confirmation_clock") != "execution":
            if self.state.position.get("stop_loss_confirmation") == "next_bar_close":
                self.state.resolution_ms = self.config.step_ms
            self.orders.finish_stop_confirmation(bar)
        if self.records.execution_equity and self.state.position is None:
            self.records.execution_equity[-1].update(cash=self.state.cash, unrealized=0.0)
        current = self.account.equity(bar.close)
        if not isfinite(current):
            raise ValueError(f"账户数值溢出：{bar.timestamp}")
        self.records.peak = max(self.records.peak, current)
        self.records.equity.append(
            {
                "timestamp": bar.timestamp + self.config.step_ms,
                "equity": current,
                "cash": self.state.cash,
                "unrealized": current
                - self.state.cash
                - (self.state.position["principal"] if self.state.position else 0),
                "drawdown_pct": (self.records.peak - current) / self.records.peak * 100,
            }
        )
        return current

    def finish(self, bars):
        if self.state.pending:
            self.state.pending[0]["status"] = "unexecuted_end"
        if self.state.pending_entry:
            self.state.pending_entry[0]["status"] = "unexecuted_end"
        if self.state.position:
            self.account.close(bars[-1].close, bars[-1], "close", "end_of_backtest")
            self.records.equity[-1].update(cash=self.state.cash, unrealized=0.0)
            if self.records.execution_equity:
                self.records.execution_equity[-1].update(cash=self.state.cash, unrealized=0.0)

    def result(self):
        report = build_report(self.config.initial_cash, self.state.cash, self.records.trades, self.records.equity)
        if any((s.get("order_type") == "stop_entry" for s in self.records.signals)):
            report["notes"].append(
                "盘中突破委托仅用前一收盘已知价位；同根入场与止损先后不明的交易已标记，并按保存的 ambiguity_policy 处理"
            )
        result = {
            "report": report,
            "trades": self.records.trades,
            "signals": self.records.signals,
            "fills": self.records.fills,
            "equity": self.records.equity,
            "levels": self.records.levels,
            "events": self.records.events,
        }
        if self.config.execution_bars is not None:
            result["execution_equity"] = self.records.execution_equity
            report["execution_close_max_drawdown_pct"] = max(
                (p["drawdown_pct"] for p in self.records.execution_equity), default=0.0
            )
            report["execution_drawdown_basis"] = "mixed_closes" if self.records.fallback_candles else "execution_closes"
            result["execution_detail"] = dict(
                detailed_candles=self.records.detailed_candles,
                fallback_candles=self.records.fallback_candles,
                profit_protection_update="execution_close"
                if self.config.intrabar_profit_protection
                else "signal_close",
            )
            report["notes"].append(
                "交易信号保持原周期；匹配的细周期行情逐根成交。细周期缺失或不匹配时回退原周期，已计数。保护仅在选定周期收盘更新，下一根生效。"
            )
        if self.records.observations:
            result["indicators"] = self.records.observations
        return result

    def run(self, bars, strategy, progress=None):
        for i, bar in enumerate(bars):
            self.past_bars.append(bar)
            if i < self.config.warmup:
                continue
            detail = self.prepare_bar(bar)
            active_entry, self.state.pending_entry = self.state.pending_entry, None
            for sub in detail or [bar]:
                active_entry = self.execute_subbar(sub, active_entry, detail)
            current = self.finish_bar(bar)
            intent, entry = self.decisions.decide(strategy, bar, current)
            self.decisions.schedule(intent, entry, bar)
            if not (detail and self.config.intrabar_profit_protection):
                self.state.resolution_ms = self.config.step_ms
                self.protection.update(bar)
            if progress and (i % 250 == 0 or i == len(bars) - 1):
                progress((i - self.config.warmup + 1) / (len(bars) - self.config.warmup))
        self.finish(bars)
        return self.result()


def simulate(
    bars,
    strategy,
    parameters,
    *,
    warmup,
    initial_cash=10000,
    position_pct=100,
    step_ms=3600000,
    progress=None,
    execution_bars=None,
    intrabar_profit_protection=False,
    execution_step_ms=None,
    record_factory=None,
):
    if intrabar_profit_protection and execution_bars is None:
        raise ValueError("盘中利润保护需要细周期成交行情")
    config = SimulationConfig(
        initial_cash, position_pct, step_ms, warmup, execution_bars, intrabar_profit_protection, execution_step_ms
    )
    return Simulation(config, parameters, record_factory).run(bars, strategy, progress)
