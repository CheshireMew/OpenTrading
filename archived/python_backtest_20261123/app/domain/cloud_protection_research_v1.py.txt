"""Historical research API, backed by the shared causal research kernel."""

from typing import Callable
from app.domain.backtest_types import Bar
from app.domain.cloud_research_kernel import simulate_research
from app.domain.cloud_research_policies import PeakPricePolicy


def simulate(
    bars: list[Bar],
    strategy,
    parameters: dict,
    *,
    warmup: int,
    initial_cash: float = 10000,
    position_pct: float = 100,
    step_ms: int = 3600000,
    progress: Callable | None = None,
    peak_price_drawdown_pct: float = 0,
) -> dict:
    return simulate_research(
        bars,
        strategy,
        parameters,
        policy=PeakPricePolicy(peak_price_drawdown_pct),
        warmup=warmup,
        initial_cash=initial_cash,
        position_pct=position_pct,
        step_ms=step_ms,
        progress=progress,
    )
