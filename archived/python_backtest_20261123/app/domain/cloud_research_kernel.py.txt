"""One historical close-decision loop for the adaptive, peak and window research APIs."""

from dataclasses import asdict
from math import isfinite
from types import MappingProxyType

from app.domain.backtest_accounting import position_equity, open_position, closed_trade, threshold_hits
from app.domain.backtest_protection import ratchet_stops
from app.domain.backtest_report import build_report
from app.domain.backtest_state import BacktestRecords, ExecutionState, SimulationConfig
from app.domain.backtest_types import Context, History, Intent
from app.domain.backtest_validation import validate_risk_intent


class ResearchAccount:
    def __init__(self, state, records, config, policy):
        self.state, self.records, self.config, self.policy = state, records, config, policy

    def close(self, price, bar, phase, reason, ambiguous=False):
        position = self.state.position
        trade = closed_trade(position, price, bar, phase, reason, ambiguous, self.config.step_ms)
        self.state.cash += position["principal"] + trade["profit"]
        self.records.trades.append(trade)
        self.records.fills.append(
            dict(
                timestamp=bar.timestamp,
                phase=phase,
                action="close",
                price=price,
                quantity=position["quantity"],
                trade_id=position["trade_id"],
                reason=reason,
            )
        )
        self.state.position = None

    def enter(self, intent, bar):
        self.state.cash, self.state.position = open_position(
            self.state.cash, self.config.position_pct, intent, bar.open, bar.timestamp, len(self.records.trades) + 1
        )
        self.policy.on_entry(self.state.position, intent, bar.open)
        position = self.state.position
        self.records.fills.append(
            dict(
                timestamp=bar.timestamp,
                phase="open",
                action=intent.action,
                price=bar.open,
                quantity=position["quantity"],
                trade_id=position["trade_id"],
                reason=intent.reason,
            )
        )

    def thresholds(self, bar, gap=False):
        position = self.state.position
        if position is None:
            return
        stopped, gained = threshold_hits(position, bar, gap)
        if stopped or gained:
            price = bar.open if gap else position["stop"] if stopped else position["target"]
            self.close(
                price,
                bar,
                "open" if gap else "intrabar",
                position.get("stop_source", "stop_loss") if stopped else "take_profit",
                stopped and gained,
            )

    def process_pending(self, bar):
        if self.state.pending is None:
            return
        signal, intent = self.state.pending
        self.state.pending = None
        position = self.state.position
        if intent.action == "close":
            if position:
                self.close(bar.open, bar, "open", "signal")
                signal["status"] = "executed"
            else:
                signal["status"] = "ignored_flat"
        elif position and position["side"] == intent.action:
            signal["status"] = "ignored_same_side"
        else:
            if position:
                self.close(bar.open, bar, "open", "reversal")
            if self.state.cash <= 0:
                signal["status"] = "ignored_nonpositive_equity"
            else:
                self.enter(intent, bar)
                signal["status"] = "executed"


class ResearchSimulation:
    def __init__(self, config, parameters, policy):
        self.config, self.policy = config, policy
        self.state, self.records = ExecutionState(config.initial_cash), BacktestRecords(config.initial_cash)
        self.account = ResearchAccount(self.state, self.records, config, policy)
        self.past = []
        self.history, self.parameters = History(self.past), MappingProxyType(dict(parameters))

    def decide(self, strategy, bar, current):
        try:
            position = MappingProxyType(dict(self.state.position)) if self.state.position else None
            result = strategy.on_bar(Context(self.history, position, self.state.cash, current, self.parameters))
            intent, distance = self.policy.decode(result, bar, self.records.observations)
            intent = Intent() if intent is None else intent
            validate_risk_intent(intent)
            return intent, distance
        except Exception as exc:
            raise ValueError(
                f"策略 {getattr(strategy, '__file__', type(strategy).__name__)}，K 线 {bar.timestamp}: {exc}"
            ) from exc

    def finish(self, bars, transition_bar=None):
        if self.state.pending:
            self.state.pending[0]["status"] = "cancelled_parameter_change" if transition_bar else "unexecuted_end"
        if transition_bar is not None:
            if transition_bar.timestamp < bars[-1].timestamp + self.config.step_ms:
                raise ValueError("Transition execution must follow the last completed candle")
            if self.state.position:
                self.account.close(transition_bar.open, transition_bar, "open", "parameter_change")
            self.records.peak = max(self.records.peak, self.state.cash)
            point = self.point(transition_bar.timestamp, self.state.cash)
            if self.records.equity[-1]["timestamp"] == point["timestamp"]:
                self.records.equity[-1] = point
            else:
                self.records.equity.append(point)
        elif self.state.position:
            self.account.close(bars[-1].close, bars[-1], "close", "end_of_backtest")
            self.records.equity[-1].update(cash=self.state.cash, unrealized=0.0)

    def point(self, timestamp, equity):
        return dict(
            timestamp=timestamp,
            equity=equity,
            cash=self.state.cash,
            unrealized=equity - self.state.cash - (self.state.position["principal"] if self.state.position else 0),
            drawdown_pct=(self.records.peak - equity) / self.records.peak * 100,
        )

    def run(self, bars, strategy, progress):
        for i, bar in enumerate(bars):
            self.past.append(bar)
            if i < self.config.warmup:
                continue
            if not self.records.equity:
                self.records.equity.append(self.point(bar.timestamp, self.state.cash))
            self.records.record_levels(bar, self.state.position)
            self.account.thresholds(bar, gap=True)
            self.account.process_pending(bar)
            self.records.record_levels(bar, self.state.position)
            self.account.thresholds(bar)
            current = position_equity(self.state.cash, self.state.position, bar.close)
            if not isfinite(current):
                raise ValueError(f"账户数值溢出：{bar.timestamp}")
            self.records.peak = max(self.records.peak, current)
            self.records.equity.append(self.point(bar.timestamp + self.config.step_ms, current))
            intent, distance = self.decide(strategy, bar, current)
            if intent.action != "hold":
                fields = asdict(intent)
                if intent.profit_protection is None:
                    fields.pop("profit_protection")
                if intent.stop_loss_distance is None:
                    fields.pop("stop_loss_distance")
                signal = dict(
                    timestamp=bar.timestamp,
                    decision_time=bar.timestamp + self.config.step_ms,
                    **fields,
                    status="pending",
                )
                self.records.signals.append(signal)
                self.state.pending = signal, intent
            ratchet_stops(self.state.position, bar, self.policy.candidate(distance))
            if progress and (i % 250 == 0 or i == len(bars) - 1):
                progress((i - self.config.warmup + 1) / (len(bars) - self.config.warmup))
        self.policy.finish(self, bars)
        records = self.records
        result = dict(
            report=build_report(self.config.initial_cash, self.state.cash, records.trades, records.equity),
            trades=records.trades,
            signals=records.signals,
            fills=records.fills,
            equity=records.equity,
            levels=records.levels,
        )
        if records.observations:
            result["indicators"] = records.observations
        return result


def simulate_research(bars, strategy, parameters, *, policy, warmup, initial_cash, position_pct, step_ms, progress):
    config = SimulationConfig(initial_cash, position_pct, step_ms, warmup)
    return ResearchSimulation(config, parameters, policy).run(bars, strategy, progress)
