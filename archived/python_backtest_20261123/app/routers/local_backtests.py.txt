from typing import Literal
from fastapi import Depends, Query, Request, HTTPException
from app.contracts.frontend import FrontendContractRouter
from app.contracts.local_backtest import (
    LocalBacktestCommand,
    LocalBacktestCatalog,
    LocalBacktestRun,
    LocalBacktestRuns,
    LocalBacktestRecords,
    LocalBacktestExport,
    LocalBacktestAction,
    LocalBacktestPreview,
)
from app.infra.executor import run_database
from app.contracts.parameter_sweep import (
    ParameterSweepCommand,
    ParameterSweepPreview,
    ParameterSweepView,
    ParameterSweepList,
    ParameterSweepAction,
    ParameterSweepEffects,
)
from app.contracts.research_results import (
    ResearchResultQuery,
    ResearchResultPage,
    ResearchResultEntry,
    ResearchResultImport,
    ResearchResultRefresh,
)
from app.contracts.backtest_diagnostics import BacktestDiagnosticSummary, BacktestTradeChanges, BacktestTradeTimeline

router = FrontendContractRouter(frontend_contract_target="research")


def local_services(request: Request):
    services = request.app.state.runtime_services.require_api().local_research
    if services is None:
        raise RuntimeError("Local research services are not initialized")
    return services


def service(services=Depends(local_services)):
    return services.backtests


def results_service(services=Depends(local_services)):
    return services.results


def sweep_service(services=Depends(local_services)):
    return services.sweeps


def diagnostic_service(services=Depends(local_services)):
    return services.diagnostics


async def invoke(call):
    try:
        return await run_database(call)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc


@router.post("/research-results/query", response_model=ResearchResultPage)
async def query_research_results(body: ResearchResultQuery, s=Depends(results_service)):
    return await invoke(lambda: s.query(**body.model_dump()))


@router.post("/local-backtests/preview", response_model=LocalBacktestPreview)
async def preview_local_backtest(body: LocalBacktestCommand, s=Depends(service)):
    return await invoke(lambda: s.preview(body))


@router.post("/research-results/refresh", response_model=ResearchResultRefresh)
async def refresh_research_results(body: ResearchResultImport, s=Depends(results_service)):
    from config.settings import BASE_DIR

    return await invoke(lambda: s.refresh(body.roots or [BASE_DIR / "docs/verification"]))


@router.get("/research-results/{entry_id}", response_model=ResearchResultEntry)
async def get_research_result(entry_id: str, s=Depends(results_service)):
    return await invoke(lambda: s.get(entry_id))

@router.get("/parameter-sweeps", response_model=ParameterSweepList)
async def list_parameter_sweeps(
    offset: int = Query(0, ge=0), limit: int = Query(20, ge=1, le=100), s=Depends(sweep_service)

    return await invoke(lambda: s.listing(offset, limit))


@router.get("/parameter-sweeps/{sweep_id}", response_model=ParameterSweepView)
async def get_parameter_sweep(
    sweep_id: str,
    offset: int = Query(0, ge=0),
    limit: int = Query(30, ge=1, le=512),
    sort: Literal["return_pct", "win_rate", "max_drawdown_pct", "total_trades"] = "return_pct",
    descending: bool = True,
    minimum_trades: int = Query(0, ge=0),
    s=Depends(sweep_service),
):
    return await invoke(lambda: s.get(sweep_id, offset, limit, sort, descending, minimum_trades))


@router.post("/parameter-sweeps/{sweep_id}/action", response_model=ParameterSweepView)
async def act_parameter_sweep(sweep_id: str, body: ParameterSweepAction, s=Depends(sweep_service)):
    batch = s
    return await invoke(lambda: batch.resume(sweep_id) if body.action == "resume" else batch.cancel(sweep_id))

@router.get("/local-backtests/{run_id}/records/{kind}", response_model=LocalBacktestRecords)
async def local_backtest_records(
    run_id: str,
    kind: Literal[
        "trades", "signals", "fills", "equity", "levels", "candles", "indicators", "events", "execution_equity"
    ],
    offset: int = Query(0, ge=0),
    limit: int = Query(500, ge=1, le=5000),
    start: int | None = None,
    end: int | None = None,
    sampled: bool = False,
    s=Depends(service),
):
    return await invoke(
        lambda: (
            s.sampled_equity(run_id, limit, kind)
            if kind in {"equity", "execution_equity"} and sampled
            else s.records(run_id, kind, offset, limit, start, end)
        )
    )
