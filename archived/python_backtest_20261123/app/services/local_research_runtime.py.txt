"""Compose the shared backtest, result, sweep and diagnostic services once per runtime."""

from dataclasses import dataclass
from pathlib import Path

from app.services.backtest_diagnostics_service import BacktestDiagnosticsService
from app.services.local_backtest_service import LocalBacktestService
from app.services.parameter_sweep_service import ParameterSweepService
from app.services.research_results_catalog import ResearchResultsCatalog


@dataclass(frozen=True)
class LocalResearchServices:
    backtests: LocalBacktestService
    results: ResearchResultsCatalog
    sweeps: ParameterSweepService
    diagnostics: BacktestDiagnosticsService


def build_local_research(database, root: Path | None = None, *, backtests=None, study_root=None):
    backtests = backtests or LocalBacktestService(database, root)
    return LocalResearchServices(
        backtests,
        ResearchResultsCatalog(backtests),
        ParameterSweepService(backtests, study_root),
        BacktestDiagnosticsService(backtests),
    )
