"""Freeze the project's inherited strategies as self-contained execution sources.

The authoring snapshots stay byte-for-byte intact. Only the source handed to
the catalog and new runs is composed; saved runs keep their own frozen source.
Metadata is evaluated from literals, never by executing strategy code in the API.
"""
import ast
from copy import deepcopy
from functools import lru_cache
from pathlib import Path

COMPOSITIONS = frozenset({
    "trend_cloud_atr_confirm_v5.py", "trend_cloud_direct_v7.py", "trend_cloud_timeframe_v6.py",
})
PARENT = "trend_cloud_parameters_v4.py"


def _literal(node, names):
    if isinstance(node, ast.Name) and node.id in names:
        return names[node.id]
    if isinstance(node, ast.Subscript):
        return _literal(node.value, names)[_literal(node.slice, names)]
    if isinstance(node, ast.Dict):
        return {_literal(k, names): _literal(v, names) for k, v in zip(node.keys, node.values)}
    if isinstance(node, (ast.List, ast.Tuple)):
        return [_literal(item, names) for item in node.elts]
    if isinstance(node, ast.Call) and isinstance(node.func, ast.Name):
        if node.func.id == "dict" and not node.args and all(k.arg is not None for k in node.keywords):
            return {k.arg: _literal(k.value, names) for k in node.keywords}
        if node.func.id == "deepcopy" and len(node.args) == 1 and not node.keywords:
            return deepcopy(_literal(node.args[0], names))
    return ast.literal_eval(node)


@lru_cache(maxsize=16)
def _compose(source: bytes, parent_source: bytes, filename: str) -> bytes:
    parent = ast.parse(parent_source.decode("utf-8-sig"), filename=PARENT)
    child = ast.parse(source.decode("utf-8-sig"), filename=filename)
    parent_metadata = next(n.value for n in parent.body if isinstance(n, ast.Assign)
                           and any(isinstance(t, ast.Name) and t.id == "STRATEGY" for t in n.targets))
    names = {"V4_SPEC": _literal(parent_metadata, {}), "OLD": _literal(parent_metadata, {})}
    body = []
    for node in parent.body:
        if isinstance(node, ast.ImportFrom) or isinstance(node, ast.Import):
            body.append(node)
        elif isinstance(node, ast.ClassDef):
            if node.name == "Calculation":
                node.name = "_InheritedCalculation"
            body.append(node)

    class InheritedNames(ast.NodeTransformer):
        def visit_Name(self, node):
            if node.id == "V4Calculation":
                node.id = "_InheritedCalculation"
            elif node.id == "AdaptiveIntent":
                node.id = "Intent"
            elif node.id == "AdaptiveResult":
                node.id = "BarResult"
            return node

    for node in child.body:
        if isinstance(node, ast.ImportFrom) and node.module == "app.domain.cloud_adaptive_types_v1":
            continue
        if isinstance(node, ast.ImportFrom) and (node.module or "").startswith("strategies."):
            if node.module != "strategies.trend_cloud_parameters_v4":
                raise ValueError("Unknown inherited strategy dependency")
            continue
        if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == "STRATEGY" for t in node.targets):
            names["STRATEGY"] = _literal(node.value, names)
            continue
        if isinstance(node, ast.Expr) and isinstance(node.value, ast.Call):
            call = node.value
            if isinstance(call.func, ast.Attribute) and call.func.attr == "update":
                target = _literal(call.func.value, names)
                for arg in call.args:
                    target.update(_literal(arg, names))
                target.update({k.arg: _literal(k.value, names) for k in call.keywords})
                continue
        body.append(InheritedNames().visit(node))
    # Put the complete static declaration last so runtime and inspection agree.
    body.append(ast.Assign(targets=[ast.Name(id="STRATEGY", ctx=ast.Store())],
                           value=ast.parse(repr(names["STRATEGY"]), mode="eval").body))
    tree = ast.fix_missing_locations(ast.Module(body=body, type_ignores=[]))
    return (ast.unparse(tree) + "\n").encode("utf-8")


def read_strategy_source(path: Path) -> bytes:
    source = path.read_bytes()
    if path.name not in COMPOSITIONS:
        return source
    return _compose(source, path.with_name(PARENT).read_bytes(), path.name)
