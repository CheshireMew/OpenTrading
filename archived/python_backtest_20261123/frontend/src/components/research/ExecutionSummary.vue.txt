<template>
  <div class="execution-summary text-sm app-muted space-y-2">
    <p>{{ tr('researchUi.signalExecutionProfitProtectionUpdatesSavedCloseBasedProtection', { p0: config.timeframe || tr('researchUi.notRecorded'), p1: executionFrame, p2: protectionFrame, p3: config.timeframe || tr('researchUi.recordedTimeframe') }) }}</p>
    <p v-if="config.execution_timeframe">{{ tr('researchUi.executionDataMatch') }} {{ summary.matched_intervals ?? (completed ? tr('researchUi.notRecorded') : tr('researchUi.unchecked')) }} {{ tr('researchUi.intervalsUsingSignalBars') }} {{ summary.fallback_intervals ?? (completed ? tr('researchUi.notRecorded') : tr('researchUi.unchecked')) }} {{ tr('researchUi.intervals') }}<span v-if="summary.coverage_pct != null">{{ tr('researchUi.covering', { p0: Number(summary.coverage_pct).toFixed(2) }) }}</span>。{{ gapPolicy === 'reject' ? tr('researchUi.stopOnMissingOrMismatchedData') : gapPolicy === 'coarse_fallback' ? tr('researchUi.allowRecordedFallbacksOnMissingOrMismatchedData') : tr('researchUi.theOriginalRecordDoesNotSpecifyHowMissingOr') }}</p>
    <p v-if="completed && provenance?.status === 'insufficient'">{{ tr('researchUi.originalEvidenceIsIncompleteYouCanViewTheOriginal', { p0: missingNames }) }}</p>
    <p v-if="config.execution_timeframe && summary.fallback_intervals">{{ tr('researchUi.executionDrawdownIncludesSignalCloseSamplesInFallbackIntervals') }}</p>
  </div>
</template>

<script setup lang="ts">
import { useI18n } from 'vue-i18n'
import { computed } from 'vue'

const { t: tr, locale } = useI18n()
const props = defineProps<{ config: Record<string, unknown>; summary: Record<string, any>; completed?: boolean }>()
const executionFrame = computed(() => 'execution_timeframe' in props.summary ? props.summary.execution_timeframe ?? tr('researchUi.notRecorded') : props.config.execution_timeframe || props.config.timeframe || tr('researchUi.notRecorded'))
const protectionFrame = computed(() => 'protection_timeframe' in props.summary ? props.summary.protection_timeframe ?? tr('researchUi.notRecorded') : (props.config.intrabar_profit_protection ? props.config.execution_timeframe : props.config.timeframe) || tr('researchUi.notRecorded'))
const gapPolicy = computed(() => 'execution_gap_policy' in props.summary ? props.summary.execution_gap_policy : props.config.execution_gap_policy)
const provenance = computed(() => props.summary.provenance as { status: string; missing: string[] } | undefined)
const names = computed<Record<string, string>>(() => ({ environment: tr('researchUi.runtimeEnvironment'), source: tr('researchUi.dataSource'), data_hash: tr('researchUi.signalDataVersion'), strategy_hash: tr('researchUi.strategyVersion'), warmup_bars: tr('researchUi.warmupRecord'), execution_bundle: tr('researchUi.completeExecutionDependencies'), execution_data_hash: tr('researchUi.executionDataVersion'), execution_alignment_hash: tr('researchUi.executionAlignmentRecord') }))
const missingNames = computed(() => provenance.value?.missing.map(k => names.value[k] || k).join('、'))
</script>
