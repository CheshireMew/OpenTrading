          <progress v-if="!terminal(selected.status)" :value="selected.progress" max="1" class="w-full" />

          <p v-if="selected.error" class="error">{{ selected.error }}</p>
          <p v-if="historyGapCount" class="error">{{ tr('researchUi.historicalDataContainsGapsNoBarsWereFabricatedOpen', { p0: historyGapCount }) }}</p>
          <details v-if="selected.metadata?.rules_summary"><summary>{{ tr('researchUi.rulesSavedForThisRun') }}</summary><p v-for="line in selected.metadata?.rules_summary as string[]" :key="line">{{ backendRuleSummary(line) }}</p></details>
          <details><summary>{{ tr('researchUi.parametersDataSourcesAndVersions') }}</summary><pre class="log">{{ JSON.stringify({ config: selected.config, source: selected.metadata?.source, version: selected.metadata?.environment }, null, 2) }}</pre></details>
        </section>
        <template v-if="selected?.status === 'completed'">
          <section ref="priceSection" class="app-panel p-5" tabindex="-1" :aria-label="tr('researchUi.priceReview')">
            <div class="flex flex-wrap items-center justify-between gap-3"><h3 class="font-semibold">{{ tr('researchUi.priceReview') }} · {{ selected.config.symbol }} · {{ selected.config.timeframe }}</h3><div class="flex gap-2"><button class="app-button-secondary" :disabled="priceLoading || chartStart <= runStart" @click="moveChart(-1)">{{ tr('chartWorkspace.previous') }}</button><button class="app-button-secondary" :disabled="priceLoading || chartEnd >= runEnd" @click="moveChart(1)">{{ tr('chartWorkspace.next') }}</button></div></div>
            <p class="text-xs app-muted mt-2">{{ date(chartStart) }} – {{ date(chartEnd) }} UTC</p>
            <p v-if="priceLevels.length" class="flex flex-wrap gap-4 text-xs mt-2"><span style="color:#e85c5c">— {{ tr('chartWorkspace.stopOrder') }}</span><span style="color:#14b88a">— {{ tr('researchUi.takeProfit') }}</span></p>
            <p v-if="priceError" role="alert" class="error">{{ priceError }} <button class="underline" @click="loadPrice">{{ tr('common.retry') }}</button></p>
            <p v-else-if="priceLoading" role="status" class="price-chart grid place-items-center app-muted">{{ tr('ui.working') }}</p>
            <div v-else-if="priceData.length" class="price-chart"><BacktestPriceChart :key="selected.id + ':' + chartStart" :data="priceData" :markers="priceMarkers" :levels="priceLevels" :indicators="priceIndicators" :logarithmic="false" :timezone-offset="0" :focus-time="focusTime" :light="theme === 'light'" /></div>
            <p v-else class="py-8 app-muted">{{ tr('researchUi.chartUnavailable') }}</p>
            <div v-if="focusedTrade" class="mt-3"><button class="app-button-secondary ml-2" @click="locateTrade(focusedTrade, true)">{{ tr('researchUi.locateExit') }}</button><TradeTimeline :key="selected.id + ':' + focusedTrade.trade_id" :run-id="selected.id" :trade-id="Number(focusedTrade.trade_id)" /></div>
          </section>
          <section class="metrics"><div v-for="[key, label] in metrics" :key="key" class="app-panel p-4"><span class="app-muted text-sm">{{ label }}</span><strong class="block text-xl mt-2">{{ number(selected.report?.[key]) }}</strong></div></section>

          <section class="app-panel p-5">
            <div class="flex flex-wrap justify-between gap-3"><h3 class="font-semibold">{{ tr('researchUi.trades') }}</h3><div class="flex gap-2"><button v-for="f in exportFormats" :key="f" class="app-button-secondary" @click="download(f)">{{ f === 'json' ? tr('researchUi.fullJson') : f === 'trades' ? tr('researchUi.tradesCsv') : tr('researchUi.equityCsv') }}</button></div></div>

            <div class="pager"><button :disabled="tradeOffset === 0" @click="tradeOffset -= 50; loadTrades()">{{ tr('chartWorkspace.previous') }}</button><span>{{ tradeOffset + trades.length }} / {{ totalTrades }}</span><button :disabled="tradeOffset + 50 >= totalTrades" @click="tradeOffset += 50; loadTrades()">{{ tr('chartWorkspace.next') }}</button></div>
          </section>
        </template>
      </div>
    </div>
  </div>

<script setup lang="ts">
import { computed, nextTick, onBeforeUnmount, onMounted, reactive, ref, watch } from 'vue'
import { useI18n } from 'vue-i18n'
import { backendText, backendRuleSummary } from '@/i18n/backendText'
import { useRoute, useRouter } from 'vue-router'
import { useTheme } from '@/composables/useTheme'
import { useRequestError } from '@/composables/useRequestError'
import TradingViewChart from '@/components/TradingViewChart.vue'
import BacktestPriceChart from '@/components/research/BacktestPriceChart.vue'
import TradeTimeline from '@/components/research/TradeTimeline.vue'
import type { TradingViewDataPoint } from '@/components/tradingViewChartPresentation'
import BacktestPreflight from '@/components/research/BacktestPreflight.vue'
import { parameterEnabled } from '@/modules/research/strategyParameters'
import { createTradePresentation } from '@/modules/research/tradePresentation'
import { localBacktestApi as api } from '@/modules/research/localBacktestApi'
import type { LocalBacktestCatalog, LocalBacktestCommand, LocalBacktestRun, LocalBacktestPreview } from '@/modules/research/generatedContracts'

const { t: tr, locale } = useI18n()
const { exitReason: reason, fillMarkers } = createTradePresentation(tr)
const { error, setError, clearError } = useRequestError(tr)
const { theme } = useTheme()
const { error: priceError, setError: setPriceError, clearError: clearPriceError } = useRequestError(tr)
type Row = Record<string, unknown>
const route = useRoute(), router = useRouter()
const catalog = ref<LocalBacktestCatalog>(), runs = ref<LocalBacktestRun[]>([]), selected = ref<LocalBacktestRun | null>(null)
const busy = ref(false), archived = ref(false)
const listOffset = ref(0), totalRuns = ref(0), tradeOffset = ref(0), totalTrades = ref(0)
const trades = ref<Row[]>([]), equity = ref<Row[]>([])
const focusTime = ref<number | null>(null)
const priceSection = ref<HTMLElement | null>(null), priceLoading = ref(false), chartStart = ref(0)
const priceData = ref<TradingViewDataPoint[]>([]), priceLevels = ref<Row[]>([]), priceIndicators = ref<Row[]>([]), priceFills = ref<Row[]>([]), focusedTrade = ref<Row | null>(null)
const runStart = computed(() => Date.parse(String(selected.value?.config.start || '')))
const runEnd = computed(() => Date.parse(String(selected.value?.config.end || '')))
const chartSteps: Record<string, number> = {'1m':60000,'5m':300000,'15m':900000,'30m':1800000,'1h':3600000,'4h':14400000,'1d':86400000}
const chartStep = computed(() => chartSteps[String(selected.value?.config.timeframe || '1h')] || 3600000)
const chartEnd = computed(() => Math.min(runEnd.value, chartStart.value + chartStep.value * 500))
const priceMarkers = computed(() => fillMarkers(priceFills.value, trades.value, focusedTrade.value?.trade_id))
let chartRequest = 0
async function loadPrice() {
  const id = selected.value?.id, start = chartStart.value, end = chartEnd.value, request = ++chartRequest
  if (!id || !Number.isFinite(start) || !Number.isFinite(end)) return
  priceLoading.value = true; clearPriceError(); priceData.value = []
  try {
    const query = { start, end: end - 1, limit: 5000 }
    const [bars, levels, indicators, fills] = await Promise.all((['candles','levels','indicators','fills'] as const).map(kind => api.records(id, kind, query)))
    if (disposed || request !== chartRequest || selected.value?.id !== id) return
    priceData.value = bars!.items.map(v => ({ time: Number(v.timestamp) / 1000, open: Number(v.open), high: Number(v.high), low: Number(v.low), close: Number(v.close) }))
    priceLevels.value = levels!.items; priceIndicators.value = indicators!.items; priceFills.value = fills!.items
  } catch (e) { if (!disposed && request === chartRequest) setPriceError(e,'ui.retryOperation') }
  finally { if (request === chartRequest) priceLoading.value = false }
}
async function moveChart(direction: number) { chartStart.value = Math.max(runStart.value, Math.min(runEnd.value - chartStep.value, chartStart.value + direction * chartStep.value * 500)); await loadPrice() }
async function locateTrade(trade: Row, exit = false) {
  focusedTrade.value = trade; focusTime.value = Number(trade[exit ? 'exit_time' : 'entry_time'])
  chartStart.value = Math.max(runStart.value, Number(trade[exit ? 'exit_time' : 'entry_time']) - chartStep.value * 25)
  await loadPrice(); await nextTick(); priceSection.value?.scrollIntoView({block:'start'}); priceSection.value?.focus({preventScroll:true})
}
const equityChart = ref<InstanceType<typeof TradingViewChart> | null>(null), drawdownChart = ref<InstanceType<typeof TradingViewChart> | null>(null)
const resultSection = ref<HTMLElement | null>(null)
watch(equity, async () => { await nextTick(); equityChart.value?.chart?.timeScale().fitContent(); drawdownChart.value?.chart?.timeScale().fitContent() }, { flush: 'post' })
const strategyLabel = (run: LocalBacktestRun) => String((run.metadata?.strategy as Row | undefined)?.name || run.config.strategy)
const runSummary = (run: LocalBacktestRun) => {
  const summary = run.metadata?.rules_summary
  return Array.isArray(summary) ? String(summary[0] ?? '') : String((run.metadata?.strategy as Row | undefined)?.description ?? '')
}
const historyGapCount = computed(() => {
  const gaps = selected.value?.metadata?.history_gaps
  return Array.isArray(gaps) ? gaps.length : 0
})
const end = new Date(); end.setUTCHours(0, 0, 0, 0)
const form = reactive({ strategy: 'ma_cross', symbol: 'BTC/USDT', timeframe: '1h', start: new Date(end.getTime() - 90 * 86400000).toISOString().slice(0, 16), end: end.toISOString().slice(0, 16), initial_cash: 10000, position_pct: 100 })
const parameters = reactive<Record<string, any>>({})
const strategy = computed(() => catalog.value?.strategies.find(s => s.id === form.strategy))
watch(catalog, value => { if (!value?.strategies.some(s => s.id === form.strategy && !s.error)) form.strategy = value?.strategies.find(s => !s.error)?.id ?? '' })
watch(strategy, s => { Object.keys(parameters).forEach(k => delete parameters[k]); for (const [k, p] of Object.entries(s?.parameters ?? {})) parameters[k] = p.default })
watch(() => form.symbol, s => { const list = catalog.value?.timeframes[s] ?? []; if (!list.includes(form.timeframe)) form.timeframe = list[0] ?? '' })
const terminal = (s: string) => ['completed', 'failed', 'cancelled', 'interrupted'].includes(s)
const status = (s: string) => ({ queued: tr('chartWorkspace.queued'), preparing: tr('researchUi.preparingData'), running: tr('researchUi.computingAndRecording'), saving: tr('researchUi.validatingAndSaving'), completed: tr('chartWorkspace.completed'), failed: tr('chartWorkspace.failed'), cancelled: tr('chartWorkspace.cancelled'), interrupted: tr('chartWorkspace.interrupted') }[s] ?? s)
const number = (v: unknown) => v == null ? '—' : Number(v).toLocaleString(locale.value, { maximumFractionDigits: 6 })
const date = (v: unknown) => new Date(typeof v === 'number' ? v : String(v)).toISOString().replace('T', ' ').replace('.000Z', '')
const metrics = computed(() => ([['return_pct', tr('researchUi.totalReturn')], ['final_equity', tr('researchUi.finalEquity')], ['profit', tr('researchUi.realizedProfitLoss')], ['max_drawdown_pct', tr('researchUi.signalCloseMaximumDrawdown')], ['total_trades', tr('web3Rank.dialog.trades')], ['win_rate', tr('chartWorkspace.winRate')], ['average_return_pct', tr('researchUi.averageTradeReturn')], ['payoff_ratio', tr('chartWorkspace.payoffRatio')], ['profit_factor', tr('researchUi.profitFactor')], ['average_duration_minutes', tr('researchUi.averageHoldingTimeMinutes')], ['breakeven', tr('researchUi.breakevenTrades')], ['ambiguous_trades', tr('researchUi.tradesWithUncertainEventOrder')]] as const))
const exportFormats = ['json', 'trades', 'equity'] as const
const line = (key: string) => equity.value.map(v => ({ time: Number(v.timestamp) / 1000, open: Number(v[key]), high: Number(v[key]), low: Number(v[key]), close: Number(v[key]) }))
const equityData = computed(() => line('equity')), drawdownData = computed(() => line('drawdown_pct'))
let timer: ReturnType<typeof setTimeout> | undefined, disposed = false, refreshBusy = false
async function safe(work: () => Promise<unknown>) { try { await work() } catch (e) { if (!disposed) setError(e, 'ui.operationFailed') } }
async function reload() { clearError(); await safe(async () => { catalog.value = await api.catalog() }); await refresh() }
async function refresh() {
  if (disposed || refreshBusy) return
  refreshBusy = true
  try { await safe(async () => {
    const page = await api.list(listOffset.value, archived.value)
    if (disposed) return
    runs.value = page.items; totalRuns.value = page.total
    const id = selected.value?.id || String(route.query.run || '')
    if (id) {
      const previous = selected.value, run = await api.get(id)
      if (!disposed && (!selected.value || selected.value.id === id)) {
        selected.value = run
        if (run.status === 'completed' && (previous?.status !== 'completed' || previous.id !== id)) await loadResults()
      }
    }
  }) } finally { refreshBusy = false }
}
async function choose(run: LocalBacktestRun) {
  ++chartRequest; focusTime.value = null; focusedTrade.value = null; priceData.value = []; selected.value = run; tradeOffset.value = 0; trades.value = []; equity.value = []
  await router.replace({ query: { ...route.query, run: run.id } })
  if (!disposed && selected.value?.id === run.id && run.status === 'completed') await loadResults()
}
const dataPreview = ref<LocalBacktestPreview>(), previewStamp = ref('')
const draftStamp = computed(() => JSON.stringify({ form, parameters }))
const command = (): LocalBacktestCommand => ({ ...form, timeframe: form.timeframe as LocalBacktestCommand['timeframe'], start: new Date(form.start + 'Z').toISOString(), end: new Date(form.end + 'Z').toISOString(), parameters: { ...parameters } })
async function previewData() { busy.value = true; clearError(); await safe(async () => { const stamp = draftStamp.value; dataPreview.value = await api.preview(command()); previewStamp.value = stamp }); busy.value = false }
async function submit() { busy.value = true; clearError(); await safe(async () => { const run = await api.create(command()); if (disposed) return; await choose(run); await refresh(); await nextTick(); resultSection.value?.scrollIntoView({ block: 'start' }); resultSection.value?.focus({ preventScroll: true }) }); busy.value = false }
async function loadTrades() { await safe(async () => { const id = selected.value!.id, offset = tradeOffset.value; const p = await api.records(id, 'trades', { offset, limit: 50 }); if (!disposed && selected.value?.id === id && tradeOffset.value === offset) { trades.value = p.items; totalTrades.value = p.total } }) }
async function loadEquity() { await safe(async () => { const id = selected.value!.id; const p = await api.records(id, 'equity', { sampled: true, limit: Math.max(300, Math.min(window.innerWidth, 2000)) }); if (!disposed && selected.value?.id === id) equity.value = p.items }) }
async function loadResults() { chartStart.value = runStart.value; await Promise.all([loadTrades(), loadEquity(), loadPrice()]); if (route.query.view === 'chart') { await nextTick(); priceSection.value?.scrollIntoView({block:'start'}) } }
async function act(action: 'cancel' | 'archive') { await safe(async () => { const run = await api[action](selected.value!.id); if (disposed) return; await choose(run); await refresh() }) }
async function download(format: typeof exportFormats[number]) { await safe(async () => { const r = await api.export(selected.value!.id, format); if (disposed) return; const url = URL.createObjectURL(new Blob([r.content], { type: r.media_type + ';charset=utf-8' })); const a = document.createElement('a'); a.href = url; a.download = r.filename; a.click(); URL.revokeObjectURL(url) }) }
async function poll() { await refresh(); if (!disposed) timer = setTimeout(poll, 2000) }
onMounted(async () => { await safe(async () => { catalog.value = await api.catalog() }); if (!disposed) await poll() })
onBeforeUnmount(() => { disposed = true; if (timer) clearTimeout(timer) })
</script>
