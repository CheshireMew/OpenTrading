<template>
  <div class="results-page">
    <div class="heading"><h1 class="text-2xl font-semibold">{{ tr('nav.researchResults') }}</h1><button class="app-button-secondary" :disabled="busy" @click="refreshImports">{{ tr('researchUi.updateLegacyIndex') }}</button></div>
    <p v-if="message" class="notice" role="status">{{ message }}</p><p v-if="error" class="error" role="alert">{{ error }}</p>
    <form class="app-panel filters" @submit.prevent="search(true)">
      <label>{{ tr('portfolioBalance.symbol') }}<input v-model.trim="symbol" :placeholder="tr('researchUi.forExampleEthUsdt')" /></label>
      <label>{{ tr('researchUi.strategyIdentifier') }}<input v-model.trim="strategy" :placeholder="tr('researchUi.leaveEmptyForAllStrategies')" /></label>
      <label>{{ tr('researchUi.strategyFamily') }}<input v-model.trim="family" :placeholder="tr('researchUi.forExampleEmaChannel')" /></label>
      <label>{{ tr('researchUi.signalTimeframe') }}<select v-model="timeframe"><option value="">{{ tr('chartWorkspace.all') }}</option><option v-for="t in frames" :key="t">{{ t }}</option></select></label>
      <label>{{ tr('researchUi.intervalStartUtc') }}<input v-model="start" type="datetime-local" /></label><label>{{ tr('researchUi.intervalEndUtc') }}<input v-model="end" type="datetime-local" /></label>
      <label>{{ tr('researchUi.executionTimeframe2') }}<select v-model="executionTimeframe"><option value="">{{ tr('chartWorkspace.all') }}</option><option value="none">{{ tr('researchUi.signalBarExecution') }}</option><option v-for="t in frames" :key="t">{{ t }}</option></select></label>
      <label>{{ tr('researchUi.protectionUpdates') }}<select v-model="protection"><option value="">{{ tr('chartWorkspace.all') }}</option><option value="signal">{{ tr('researchUi.signalBarClose') }}</option><option value="execution">{{ tr('researchUi.executionBarClose') }}</option></select></label>
      <label class="parameter-filter">{{ tr('researchUi.exactParameterMatchJson') }}<input v-model="parameters" :placeholder="tr('researchUi.forExampleCenterLength55WidthLength31')" /></label>
      <label>{{ tr('equities.sort') }}<select v-model="sort"><option value="return_pct">{{ tr('researchUi.returnHighToLow') }}</option><option value="win_rate">{{ tr('researchUi.winRateHighToLow') }}</option><option value="max_drawdown_pct">{{ tr('researchUi.drawdownLowToHigh') }}</option><option value="total_trades">{{ tr('researchUi.tradesMostFirst') }}</option></select></label>
      <label>{{ tr('researchUi.minimumTrades') }}<input v-model.number="minimumTrades" type="number" min="0" step="1" /></label>
      <label class="checkbox"><input v-model="includeArchived" type="checkbox" />{{ tr('researchUi.includeArchivedResults') }}</label><button class="app-button-primary" :disabled="busy" type="submit">{{ busy ? tr('researchUi.loading') : tr('researchUi.queryExistingResults') }}</button>
    </form>
    <div class="scope"><strong>{{ reference ? tr('researchUi.rankUnderTheSameConditions') : tr('researchUi.rankHistoricalRecords') }}</strong><span v-if="reference">{{ tr('researchUi.reference', { p0: reference.label }) }}</span><button v-if="reference" class="app-button-secondary" :disabled="busy" @click="reference = null; search(true)">{{ tr('researchUi.returnToHistoricalRanking') }}</button><p class="app-muted">{{ page ? tr('researchUi.resultsRankingNote') : '' }}</p></div>
    <section class="app-panel">
      <div class="table-scroll"><table><thead><tr><th>{{ tr('researchUi.resultSource') }}</th><th>{{ tr('researchUi.assetTimeframe') }}</th><th>{{ tr('researchUi.testIntervalUtc') }}</th><th>{{ tr('dca.roi') }}</th><th>{{ tr('portfolioBalance.maxDrawdown') }}</th><th>{{ tr('researchUi.winRateTrades') }}</th><th>{{ tr('researchUi.versionEvidence') }}</th></tr></thead>
        <tbody><tr v-for="item in page?.items" :key="item.id" :class="{ selected: selected?.id === item.id }"><td><button class="entry-link" @click="select(item)">{{ item.label }}</button><small>{{ item.source_type === 'native' ? tr('researchUi.localRun') : tr('researchUi.legacyResearchSummary') }}{{ item.archived ? tr('researchUi.archived') : '' }} · {{ item.family || tr('researchUi.unregisteredFamily') }}</small></td><td>{{ item.config.symbol || tr('researchUi.notRecorded') }}<small>{{ item.config.timeframe || tr('researchUi.unknownTimeframe') }} · {{ item.config.execution_timeframe ? tr('researchUi.execution', { p0: item.config.execution_timeframe }) : tr('researchUi.seeDetailsForExecutionTimeframe') }}</small></td><td>{{ date(item.config.start) }}<small>{{ tr('researchUi.to3', { p0: date(item.config.end) }) }}</small></td><td>{{ percent(item.report.return_pct) }}</td><td>{{ percent(item.report.max_drawdown_pct) }}</td><td>{{ percent(item.report.win_rate) }}<small>{{ tr('researchUi.trades2', { p0: item.report.total_trades ?? '—' }) }}</small></td><td><small v-if="item.run_id && !item.run_available" class="text-amber-700 dark:text-amber-300">{{ tr('researchUi.runUnavailable') }}</small>{{ item.cohort ? tr('researchUi.sameConditionFilteringAvailable') : tr('researchUi.incompleteEvidence') }}</td></tr></tbody>
      </table></div><p v-if="page && !page.items.length" class="app-muted py-5">{{ tr('researchUi.noMatchingResultsUpdateTheIndexWhenFirstUsing') }}</p>
      <div class="pager"><button :disabled="busy || offset === 0" @click="offset -= 30; search()">{{ tr('chartWorkspace.previous') }}</button><span>{{ page?.total ? offset + 1 : 0 }}–{{ offset + (page?.items.length ?? 0) }} / {{ page?.total ?? 0 }}</span><button :disabled="busy || offset + 30 >= (page?.total ?? 0)" @click="offset += 30; search()">{{ tr('chartWorkspace.next') }}</button></div>
    </section>
    <section v-if="selected" ref="detailSection" tabindex="-1" class="app-panel detail">
      <div class="heading"><h2 class="text-lg font-semibold">{{ selected.label }}</h2><div class="actions"><RouterLink v-if="selected.run_id && selected.run_available" :to="{ path: '/research/backtests', query: { run: selected.run_id } }" class="app-button-primary">{{ tr('researchUi.openBacktestDetails') }}</RouterLink><button class="app-button-secondary" :disabled="busy || !selected.cohort" @click="compareConditions">{{ tr('researchUi.viewSameConditionResults') }}</button><button class="app-button-secondary" @click="download">{{ tr('researchUi.exportThisRecord') }}</button></div></div>
      <div v-if="selected.run_id && !selected.run_available" role="status" class="notice"><p>{{ tr('researchUi.runUnavailableHint') }}</p><button class="app-button-secondary mt-3" :disabled="busy" @click="select(selected)">{{ tr('researchUi.recheckRun') }}</button></div>
      <p v-if="selected.context_note" class="app-muted my-3">{{ selected.context_note }}</p>
      <ExecutionSummary :config="selected.config" :summary="selected.execution_summary" completed />
      <p v-if="selected.missing.length" class="notice">{{ tr('researchUi.missingEvidencePreventsConfirmationOfTheSameConditions', { p0: selected.missing.map(name => evidenceNames[name] || name).join('、') }) }}</p>
      <details class="my-3" open><summary>{{ tr('researchUi.savedConfigurationAndParameters') }}</summary><pre>{{ JSON.stringify(selected.config, null, 2) }}</pre></details>
      <h3 class="font-semibold mt-4">{{ tr('researchUi.sourceLocation') }}</h3><p class="app-muted text-sm my-2">{{ tr('researchUi.checksumsVerifyFilesAtImportTimeUpdateTheIndex') }}</p>
      <div v-for="(source, i) in selected.sources" :key="i" class="source"><p>{{ source.path }} · {{ source.pointer }}</p><small>{{ source.exists ? (source.unchanged === false ? tr('researchUi.theFileHasChangedImportItAgain') : source.unchanged === true ? tr('researchUi.verifiedContentUnchanged') : tr('researchUi.fileExists')) : tr('researchUi.sourceFileMissing') }}</small><code v-if="source.sha256_at_import">SHA-256 {{ source.sha256_at_import }}</code></div>
    </section>
  </div>
</template>

<script setup lang="ts">
import { useI18n } from 'vue-i18n'
import { useRequestError, LocalizedRequestError } from '@/composables/useRequestError'
import { computed } from 'vue'
import { nextTick, onMounted, ref } from 'vue'
import { RouterLink } from 'vue-router'
import ExecutionSummary from '@/components/research/ExecutionSummary.vue'
import { researchResultsApi as api } from '@/modules/research/researchResultsApi'
import type { ResearchResultEntry, ResearchResultPage, ResearchResultQuery } from '@/modules/research/generatedContracts'

const { t: tr, locale } = useI18n()
const { error, setError, clearError } = useRequestError(tr)

const detailSection = ref<HTMLElement | null>(null)
const frames = ['1m', '5m', '15m', '30m', '1h', '4h', '1d']
const evidenceNames = computed<Record<string, string>>(() => ({ 'config.strategy': tr('chartWorkspace.strategy'), 'config.symbol': tr('portfolioBalance.symbol'), 'config.timeframe': tr('researchUi.signalTimeframe'), 'config.start': tr('researchUi.start'), 'config.end': tr('researchUi.end'), 'config.initial_cash': tr('chartWorkspace.initialCapital'), 'config.position_pct': tr('researchUi.positionSize'), 'config.execution_timeframe': tr('researchUi.executionTimeframe2'), 'config.intrabar_profit_protection': tr('researchUi.protectionUpdateMode'), 'config.data_gap_policy': tr('researchUi.signalDataGapPolicy'), 'config.execution_gap_policy': tr('researchUi.executionDataGapPolicy'), 'config.parameters': tr('researchUi.fullParameters'), 'metadata.source': tr('researchUi.dataSource'), 'metadata.environment': tr('researchUi.runtimeEnvironment'), 'metadata.strategy_hash': tr('researchUi.strategyVersion'), 'metadata.data_hash': tr('researchUi.signalDataVersion'), 'metadata.execution_data_hash': tr('researchUi.executionDataVersion'), 'metadata.execution_alignment_hash': tr('researchUi.executionAlignmentEvidence'), 'metadata.execution_bundle': tr('researchUi.completeExecutionDependencies'), 'metadata.warmup_bars': tr('researchUi.warmupSettings') }))
const symbol = ref(''), strategy = ref(''), family = ref(''), timeframe = ref(''), start = ref(''), end = ref('')
const executionTimeframe = ref(''), protection = ref(''), parameters = ref(''), minimumTrades = ref(0), includeArchived = ref(false)
const sort = ref<NonNullable<ResearchResultQuery['sort']>>('return_pct'), offset = ref(0), busy = ref(false), refreshOutcome = ref<Awaited<ReturnType<typeof api.refresh>> | null>(null)
const message = computed(() => {
  const r = refreshOutcome.value
  if (!r) return ''
  return tr('researchUi.indexedLocalRunsAndProcessedLegacyReferences', {
    p0: r.native_runs, p1: r.imported_references,
    p2: r.issues.length ? tr('researchUi.filesNeedReview', {
      p0: r.issues.length, p1: r.issues.slice(0, 5).map(i => `${i.path}: ${i.issue}`).join('; '),
    }) : tr('researchUi.sourceCheckComplete'),
  })
})
const page = ref<ResearchResultPage | null>(null), selected = ref<ResearchResultEntry | null>(null), reference = ref<ResearchResultEntry | null>(null)
const percent = (value: unknown) => typeof value === 'number' ? `${value.toLocaleString(locale.value, { maximumFractionDigits: 2 })}%` : '—'
const date = (value: unknown) => typeof value === 'string' ? value.replace('T', ' ').replace('+00:00', '').replace('Z', '') : tr('researchUi.notRecorded')
async function guard(task: () => Promise<void>) { busy.value = true; clearError(); try { await task() } catch (e) { setError(e, 'ui.operationFailed') } finally { busy.value = false } }
async function search(reset = false) { await guard(async () => {
  const params: unknown = parameters.value.trim() ? JSON.parse(parameters.value) : {}
  if (!params || Array.isArray(params) || typeof params !== 'object') throw new LocalizedRequestError('researchUi.parametersMustBeAJsonObjectForExampleCenter')
  if (reset) offset.value = 0
  page.value = await api.query({ symbol: symbol.value || null, strategy: strategy.value || null, family: family.value || null,
    timeframe: timeframe.value || null, start: start.value ? new Date(`${start.value}Z`).toISOString() : null, end: end.value ? new Date(`${end.value}Z`).toISOString() : null,
    execution_timeframe: executionTimeframe.value || null, intrabar_profit_protection: protection.value ? protection.value === 'execution' : null,
    parameters: params as Record<string, unknown>, reference_id: reference.value?.id ?? null, sort: sort.value,
    descending: sort.value !== 'max_drawdown_pct', minimum_trades: minimumTrades.value, include_archived: includeArchived.value, offset: offset.value, limit: 30 })
}) }
async function select(item: ResearchResultEntry) { await guard(async () => { selected.value = await api.get(item.id); await nextTick(); detailSection.value?.scrollIntoView({ block: 'start' }); detailSection.value?.focus({ preventScroll: true }) }) }
async function compareConditions() { reference.value = selected.value; await search(true) }
async function refreshImports() {
  await guard(async () => { refreshOutcome.value = await api.refresh() })
  if (!error.value) await search(true)
}
function download() { const url = URL.createObjectURL(new Blob([JSON.stringify(selected.value, null, 2)], { type: 'application/json;charset=utf-8' })); const a = document.createElement('a'); a.href = url; a.download = `${selected.value?.id.replace(':', '-')}.json`; a.click(); URL.revokeObjectURL(url) }
onMounted(() => search())
</script>
