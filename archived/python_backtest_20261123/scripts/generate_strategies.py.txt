"""Expand shared authoring fragments into self-contained, byte-stable strategy snapshots."""

import argparse
import hashlib
import json
from pathlib import Path
import re

ROOT = Path(__file__).resolve().parents[1]
AUTHORING = ROOT / "strategies/authoring"
INCLUDE = re.compile(rb"^@@include:([a-z][a-z0-9_]*)@@\r?\n", re.MULTILINE)


def rendered_strategies(authoring=AUTHORING):
    manifest = json.loads((authoring / "manifest.json").read_text(encoding="utf-8"))
    if manifest.get("version") != 1:
        raise ValueError("Unknown strategy authoring format")
    results = {}
    for name, spec in manifest["strategies"].items():
        if not re.fullmatch(r"[a-z][a-z0-9_]*\.py", name) or spec["template"] != name + ".in":
            raise ValueError("Invalid strategy output or template name")
        source = (authoring / "templates" / spec["template"]).read_bytes()

        def include(match):
            return (authoring / "shared" / (match[1].decode("ascii") + ".inc")).read_bytes()

        source = INCLUDE.sub(include, source)
        if b"@@include:" in source:
            raise ValueError(f"Unresolved strategy include: {name}")
        if hashlib.sha256(source).hexdigest() != spec["sha256"]:
            raise ValueError(f"Frozen strategy changed: {name}; create a new version instead of rewriting its snapshot")
        compile(source, name, "exec")
        results[name] = source
    return results


def generate(*, check, authoring=AUTHORING, destination=ROOT / "strategies"):
    rendered = rendered_strategies(authoring)
    changed = [
        name
        for name, source in rendered.items()
        if not (destination / name).is_file() or (destination / name).read_bytes() != source
    ]
    if not check:
        destination.mkdir(parents=True, exist_ok=True)
        for name in changed:
            (destination / name).write_bytes(rendered[name])
    return dict(strategies=len(rendered), changed=changed, check=check, valid=not changed if check else True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--check", action="store_true", help="Verify outputs without writing files")
    mode.add_argument("--write", action="store_true", help="Regenerate standalone strategy snapshots")
    args = parser.parse_args()
    result = generate(check=args.check)
    print(json.dumps(result, ensure_ascii=False))
    return 0 if result["valid"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
