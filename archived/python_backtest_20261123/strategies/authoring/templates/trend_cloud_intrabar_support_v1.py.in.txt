class KnownPriceLevels:
    """The prior context study's hourly pivots/day extremes, with no momentum gate."""
    def __init__(self):
        self.hour = CompletedBars(HOUR)
        self.day = CompletedBars(24*HOUR)
        self.hour_history = deque(maxlen=5)
        self.pivots = deque()
        self.previous_day = None

    def known_levels(self, at):
        cutoff = at-120*HOUR
        while self.pivots and self.pivots[0]['known_at'] < cutoff:
            self.pivots.popleft()
        levels = [dict(p) for p in self.pivots if p['known_at'] <= at]
        if self.previous_day and self.previous_day['known_at'] <= at:
            levels.extend(dict(price=self.previous_day[k], known_at=self.previous_day['known_at'], kind='previous_day_'+k)
                          for k in ('high', 'low'))
        return levels

    def update(self, bar):
        h = self.hour.update(bar)
        if h:
            self.hour_history.append(h)
            if len(self.hour_history) == 5:
                window = list(self.hour_history)
                pivot = window[2]
                others = window[:2]+window[3:]
                if all(pivot['high'] > x['high'] for x in others):
                    self.pivots.append(dict(price=pivot['high'], known_at=h['known_at'], formed_at=pivot['timestamp'], kind='hour_high'))
                if all(pivot['low'] < x['low'] for x in others):
                    self.pivots.append(dict(price=pivot['low'], known_at=h['known_at'], formed_at=pivot['timestamp'], kind='hour_low'))
        day = self.day.update(bar)
        if day:
            self.previous_day = day


def location_checks(levels, bar, width):
    """Levels were already available when this signal candle opened."""
    prices = [p['price'] for p in levels if p['known_at'] <= bar.timestamp]
    support = max((p for p in prices if p <= bar.close), default=None)
    resistance = min((p for p in prices if p >= bar.close), default=None)
    long_ok = bool(width > 0 and support is not None and 0 < bar.close-support <= width
                   and bar.low <= support+.25*width)
    short_ok = bool(width > 0 and resistance is not None and 0 < resistance-bar.close <= width
                    and bar.high >= resistance-.25*width)
    return support, resistance, long_ok, short_ok
