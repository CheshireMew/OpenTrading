"""The three research APIs keep every historical output while sharing one implementation."""

import random
from pathlib import Path
from types import ModuleType, SimpleNamespace

import pytest

from app.domain.local_backtest import Bar, BarResult, Intent, ProfitProtection
from app.domain.cloud_adaptive_types_v1 import AdaptiveIntent, AdaptiveResult
from app.domain import cloud_adaptive_research_v1, cloud_protection_research_v1, cloud_window_research_v1


def historical(name):
    path = Path(__file__).parent / "fixtures/research_engines_v1" / (name + ".py")
    module = ModuleType("historical_" + name)
    exec(compile(path.read_bytes(), str(path), "exec"), module.__dict__)
    return module


def strategy(adaptive):
    def on_bar(context):
        i = len(context.history)
        action = "long" if i % 8 == 1 else "short" if i % 8 == 5 else "hold"
        risk = dict(stop_loss_pct=4, profit_protection=ProfitProtection(2, 30))
        if adaptive:
            return AdaptiveResult(
                AdaptiveIntent(action, activation_distance=1.5, **risk),
                trailing_distance=2.0,
                record_observation=i % 2 == 0,
                values={"count": i},
            )
        return BarResult(Intent(action, **risk), values={"count": i})

    return SimpleNamespace(on_bar=on_bar)


@pytest.mark.parametrize("seed", [3, 12, 24])
@pytest.mark.parametrize(
    "api,options,adaptive",
    [
        (cloud_adaptive_research_v1, {}, False),
        (cloud_adaptive_research_v1, {}, True),
        (cloud_protection_research_v1, {"peak_price_drawdown_pct": 0}, False),
        (cloud_protection_research_v1, {"peak_price_drawdown_pct": 2}, False),
        (cloud_window_research_v1, {}, False),
        (cloud_window_research_v1, {"transition_bar": Bar(90000, 95, 99, 94, 96, 10)}, False),
    ],
)
def test_research_result_parity(seed, api, options, adaptive):
    rng, bars, price = random.Random(seed), [], 100.0
    for i in range(90):
        close = max(25.0, price + rng.uniform(-4, 4))
        bars.append(Bar(i * 1000, price, max(price, close) + 2, min(price, close) - 2, close, 10))
        price = close
    old = historical(api.__name__.split(".")[-1])
    kwargs = dict(warmup=0, position_pct=75, step_ms=1000, **options)
    assert api.simulate(bars, strategy(adaptive), {}, **kwargs) == old.simulate(bars, strategy(adaptive), {}, **kwargs)
