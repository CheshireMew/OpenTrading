import json

import pytest

from scripts.generate_strategies import AUTHORING, ROOT, generate, rendered_strategies


def test_every_strategy_is_generated_and_keeps_its_historical_bytes():
    rendered = rendered_strategies()
    assert set(rendered) == {path.name for path in (ROOT / "strategies").glob("*.py")}
    assert all((ROOT / "strategies" / name).read_bytes() == source for name, source in rendered.items())
    assert generate(check=True)["valid"]
    manifest = json.loads((AUTHORING / "manifest.json").read_text(encoding="utf-8"))
    assert len(manifest["strategies"]) == 24
    assert len(list((AUTHORING / "shared").glob("*.inc"))) >= 1


def test_standalone_outputs_can_be_rebuilt_without_runtime_template_dependencies(tmp_path):
    result = generate(check=False, destination=tmp_path)
    assert len(result["changed"]) == 24
    assert generate(check=True, destination=tmp_path)["valid"]
    for path in tmp_path.glob("*.py"):
        compile(path.read_bytes(), str(path), "exec")
        assert b"authoring" not in path.read_bytes()


def test_changed_shared_source_cannot_silently_rewrite_historical_strategy(tmp_path):
    import shutil

    shutil.copytree(AUTHORING, tmp_path / "authoring")
    fragment = next((tmp_path / "authoring/shared").glob("*.inc"))
    fragment.write_bytes(fragment.read_bytes() + b"\n# unexpected change\n")
    with pytest.raises(ValueError, match="Frozen strategy changed"):
        rendered_strategies(tmp_path / "authoring")
