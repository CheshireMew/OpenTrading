def test_null_quote_timestamps_do_not_prevent_loading_or_writing_the_ledger(ledger_service):
    from app.contracts.portfolios import PortfolioCollectionSaveCommand
    collection = ledger_service.get_collection()
    asset = collection.snapshot.portfolios[0].assets[0]
    asset.priceObservedAt = asset.priceValidUntil = None
    ledger_service.save_collection(PortfolioCollectionSaveCommand(
        expectedRevision=collection.revision, snapshot=collection.snapshot,
    ))
    ledger_service.create_transaction("core", _transaction("deposit", cashAmount=1000))
    ledger_service.create_transaction("core", _transaction("buy", symbol="BTC", quantity=1, unitPrice=100))
    ledger = ledger_service.get_ledger("core")
    assert len(ledger.transactions) == 2
    assert ledger.summary.positions[0].marketPrice is None
    assert ledger.summary.totalEquity is None


@pytest.mark.asyncio
async def test_report_saves_populated_research_and_returns_explicit_utc(installed_database_runtime):
    research = ResearchRepository(installed_database_runtime)
    watchlist = research.create_watchlist({"name": "Audit regression"})
    research.create_watch_item({"watchlist_id": watchlist["id"], "symbol": "BTC/USDT", "thesis": "Study"})
    response = ResearchWatchlistResponse.model_validate(watchlist)
    assert response.createdAt.tzinfo is not None
    assert response.createdAt.utcoffset().total_seconds() == 0
    service = ResearchReportService(
        repository=ResearchReportRepository(installed_database_runtime), research_repository=research,
        signal_repository=TradeSignalRepository(installed_database_runtime), portfolio_service=object(),
        data_health_service=StubHealth(), asset_research_service=object(),
    )
    report = await service.generate(ResearchReportGenerateCommand(title="Populated research", reportType="daily_brief"))
    restored = service.get_report(report.id)
    assert restored.dataSnapshot == report.dataSnapshot
    events = restored.dataSnapshot["research"]["recentEvents"]
    assert len(events) == 2
    assert all(datetime.fromisoformat(event["occurredAt"]).utcoffset().total_seconds() == 0 for event in events)
    assert restored.createdAt.tzinfo == timezone.utc


def module(source, name):
    result = ModuleType(name)
    result.__file__ = name
    exec(compile(source, name, "exec"), result.__dict__)
    return result


@pytest.mark.parametrize("name,updates", [
    ("trend_cloud_atr_confirm_v5.py", {}),
    ("trend_cloud_timeframe_v6.py", {"atr_stop_mult": 2, "reverse_confirm_bars": 2}),
    ("trend_cloud_direct_v7.py", {}),
    ("trend_cloud_direct_v7.py", {"adaptive_center": True, "scaled_risk": True, "atr_trailing": True, "efficiency_gate": True}),
])
def test_inherited_strategies_are_inspectable_standalone_and_preserve_trades(name, updates):
    path = Path(__file__).parents[1] / "strategies" / name
    original = path.read_bytes()
    composed = read_strategy_source(path)
    metadata = inspect_strategy(composed, name)
    parameters = validate_parameters(metadata, updates)
    assert name in COMPOSITIONS
    assert path.read_bytes() == original
    assert b"from strategies." not in composed
    assert b"cloud_adaptive_types" not in composed
    old, new = module(original, name), module(composed, "frozen_" + name)
    assert old.STRATEGY == new.STRATEGY
    rng, bars, price = random.Random(58), [], 100.0
    for i in range(900):
        close = max(25.0, price + rng.uniform(-3, 3))
        bars.append(Bar(i * 900000, price, max(price, close) + .2, min(price, close) - .2, close, 10))
        price = close
    options = dict(warmup=200, step_ms=900000)
    expected = (simulate_adaptive if "direct" in name else simulate)(bars, old, parameters, **options)
    actual = simulate(bars, new, parameters, **options)
    keys = ("side", "entry_time", "exit_time", "entry_price", "exit_price", "profit", "exit_reason")
    assert actual["trades"]
    assert [[row[k] for k in keys] for row in actual["trades"]] == [[row[k] for k in keys] for row in expected["trades"]]
    assert actual["indicators"] == expected["indicators"]
